[3362. 零数组变换 III](https://leetcode.cn/problems/zero-array-transformation-iii/)

- **排序查询**：
    - 将 `queries` 按照起始位置排序，以便顺序处理相关查询。
- **使用优先队列**：
    - 利用最大堆 (`priority_queue`) 存储当前有效的查询条件。
- **操作计数**：
    - 使用 `operations` 记录已执行的移除操作数量。
    - `deltaArray` 用于跟踪操作增量，帮助动态调整操作数量。
- **遍历 `nums`**：
    - 对每个元素执行以下操作：
        - 更新 `operations`。
        - 将与当前索引相同的查询添加到堆中。
        - 从堆中弹出元素，增加 `operations`，直到条件满足。
- **条件检查**：
    - 如果当前操作数量小于当前元素值，返回 `-1`，表示无法满足条件。
- **返回结果**：
    - 返回堆中剩余的查询条件数量，表示可以移除的最大元素数量。
```cpp
int maxRemoval(vector<int>& nums, vector<vector<int>>& queries) {
    // 按查询的起始位置排序
    sort(queries.begin(), queries.end(), [](const vector<int>& a, const vector<int>& b) {
        return a[0] < b[0];
    });
    
    priority_queue<int> heap; // 最大堆
    vector<int> deltaArray(nums.size() + 1, 0); // 操作增量数组
    int operations = 0; // 当前操作数量
    int j = 0; // 查询索引

    for (int i = 0; i < nums.size(); ++i) {
        operations += deltaArray[i]; // 更新当前操作数量

        // 将所有与当前索引相同的查询加入堆
        while (j < queries.size() && queries[j][0] == i) {
            heap.push(queries[j][1]);
            ++j;
        }

        // 从堆中弹出元素并增加操作数量，直到条件满足
        while (operations < nums[i] && !heap.empty() && heap.top() >= i) {
            operations += 1;
            deltaArray[heap.top() + 1] -= 1; // 更新增量数组
            heap.pop();
        }

        // 检查是否满足条件
        if (operations < nums[i]) {
            return -1; // 无法满足条件
        }
    }

    return heap.size(); // 返回可移除的最大元素数量
}
```

##### 双指针
- **初始化**：
    - 计算 `nums` 的长度 `len` 和操作计数 `opcnt`。
    - 使用嵌套的 `unordered_map` (`mp`) 来存储每个查询的范围和对应的操作数量。
- **处理查询**：
    - 遍历每个查询，将查询的左边界和右边界计算并存入 `mp` 中，同时增加操作计数 `opcnt`。
- **操作数组**：
    - 定义 `opArr` 用于记录每个位置的操作增量。
- **遍历 `nums`**：
    - 对每个元素进行以下操作：
        - 更新 `op_total`，表示当前的总操作数量。
        - 计算当前元素的有效值 `num`，即 `nums[i]` 减去已执行的操作。
- **处理操作**：
    - 使用双指针 `j` 和 `k` 来查找可以应用的操作：
        - 查找范围内的操作数量，并根据 `num` 更新 `opArr` 和 `mp`。
        - 如果在范围内找到足够的操作，则更新操作数量并退出循环；否则，继续查找。
- **条件检查**：
    - 如果在处理后 `num` 仍大于零，返回 `-1`，表示无法满足条件。
- **返回结果**：
    - 返回剩余的操作计数 `opcnt`，表示可以移除的最大元素数量。
```cpp
int maxRemoval(vector<int>& nums, vector<vector<int>>& queries) {
    int len = nums.size(), opcnt = 0;
    unordered_map<int, unordered_map<int, int>> mp;

    // 处理查询，将其存储到哈希表中
    for (auto& q : queries) {
        int left = q[0], right = q[1] - left;
        mp[left][right]++;
        opcnt++;
    }
    
    vector<int> opArr(len + 1, 0); // 操作增量数组
    int op_total = 0; // 当前操作总数

    // 遍历 nums
    for (int i = 0; i < len; i++) {
        op_total += opArr[i]; // 更新当前总操作数量
        int num = nums[i] - op_total; // 计算当前有效值

        if (num > 0) {
            int j = 0, k = len - 1, offset = 0;

            // 查找可以应用的操作
            while (k >= (i - j)) {
                if (mp[j][k]) {
                    int mtmp = mp[j][k];
                    if (mtmp >= num) {
                        opArr[j + k + 1] -= num; // 更新操作增量
                        op_total += num; // 增加操作总数
                        mp[j][k] -= num; // 减少可用操作
                        opcnt -= num; // 更新操作计数
                        num = 0; // 操作完成
                        break;
                    } else {
                        num -= mtmp; // 减去已用操作
                        op_total += mtmp; // 增加操作总数
                        opArr[j + k + 1] -= mtmp; // 更新操作增量
                        opcnt -= mtmp; // 更新操作计数
                        mp[j][k] = 0; // 重置操作
                    }
                }

                // 更新指针
                if (j == i) {
                    j = 0;
                    offset++;
                    k = len - 1 - offset;
                } else {
                    j++;
                    k--;
                }
            }
        }

        // 检查条件
        if (num > 0) return -1; // 无法满足条件
    }

    return opcnt; // 返回可移除的最大元素数量
}
```

##### 动态调整移除操作数量

- **排序查询**：
    - 使用 `ranges::sort` 对 `queries` 进行排序，以便按顺序处理每个查询。
- **初始化变量**：
    - `n` 为 `nums` 的长度，`m` 为 `queries` 的长度。
    - `sumD` 用于记录当前的操作总数，`j` 用于遍历查询。
    - `diff` 数组用于追踪操作增量。
- **优先队列**：
    - 使用最大堆（`priority_queue`）来存储当前有效的查询条件。
- **遍历 `nums`**：
    - 对每个元素 `nums[i]`：
        - 更新 `sumD`，计算当前操作总数。
        - 将所有有效的查询条件（满足 `queries[j][0] <= i`）加入堆。
        - 从堆中弹出元素，增加操作，直到满足条件或堆为空。
- **条件检查**：
    - 如果在处理某个元素后 `sumD` 仍小于该元素的值，返回 `-1`，表示无法满足条件。
- **返回结果**：
    - 返回堆中剩余的查询条件数量，表示可以移除的最大元素数量。

```cpp
int maxRemoval(vector<int>& nums, vector<vector<int>>& queries) {
    // 对查询进行排序
    ranges::sort(queries);
    
    int n = nums.size(); // nums 的长度
    int m = queries.size(); // queries 的长度
    int sumD = 0; // 当前操作总数
    int j = 0; // 查询索引
    vector<int> diff(n + 1, 0); // 操作增量数组
    priority_queue<int> pq; // 最大堆，用于存储查询条件

    // 遍历 nums
    for (int i = 0; i < n; i++) {
        sumD += diff[i]; // 更新当前操作总数

        // 将所有满足条件的查询加入堆
        while (j < m && queries[j][0] <= i) {
            pq.push(queries[j++][1]);
        }

        // 从堆中弹出元素并增加操作，直到满足条件
        while (sumD < nums[i] && !pq.empty() && i <= pq.top()) {
            sumD++;
            diff[pq.top() + 1]--; // 更新操作增量
            pq.pop(); // 弹出堆顶元素
        }

        // 检查是否满足条件
        if (sumD < nums[i]) return -1; // 无法满足条件
    }

    return pq.size(); // 返回可移除的最大元素数量
}
```
