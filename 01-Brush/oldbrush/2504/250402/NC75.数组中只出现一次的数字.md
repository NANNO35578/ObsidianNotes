- 按位异或
```cpp
    vector<int> FindNumsAppearOnce(vector<int>& nums) {
        // write code here
        int aXb = 0;
        for (int i : nums) {
            aXb ^= i;
        }
        //? to separate nums
        int spMark = aXb & (-aXb);
        int a = 0;
        for (int i : nums) {
            if (spMark & i) {
                a ^= i;
            }
        }
        return vector<int>{ (aXb ^ a),a};
    }
```
