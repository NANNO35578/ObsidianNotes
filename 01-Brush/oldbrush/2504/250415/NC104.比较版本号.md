[比较版本号](https://www.nowcoder.com/profile/386104436/codeBookDetail?submissionId=457525443)
- - **使用 `stringstream`**：将版本号转换为流，逐个读取以 `.` 分隔的部分。
- **比较每个部分**：将每个部分转换为整数进行比较，处理空部分时默认值为 0。
- **循环处理**：使用 `getline` 逐个处理每个版本号的部分，直到两个版本号的所有部分都被处理完。
```cpp
    int compare(string version1, string version2) {
        // write code here
        // 用空字符串替换点号方便处理
        stringstream ss1(version1), ss2(version2);
        string token1, token2;

        while (true) {
            // 将每个部分转为整数，方便比较
            bool hasToken1 = static_cast<bool>(getline(ss1, token1, '.'));
            bool hasToken2 = static_cast<bool>(getline(ss2, token2, '.'));
            if (!hasToken1 && !hasToken2) return 0;

            // 将每个部分转为整数，方便比较
            int v1 = hasToken1 ? stoi(token1) : 0;
            int v2 = hasToken2 ? stoi(token2) : 0;

            if (v1 < v2) return -1;
            if (v1 > v2) return 1;

            // 清空 token 以便准备下一轮
            token1.clear();
            token2.clear();
        }

        return 0;  // 如果完全相等
    }
```

    int compare(string version1, string version2) {
        int n1 = version1.size();
        int n2 = version2.size();
        int i = 0, j = 0;
        //直到某个字符串结束
        while(i < n1 || j < n2){
            long long num1 = 0;
            //从下一个点前截取数字
            while(i < n1 && version1[i] != '.'){
                num1 = num1 * 10 + (version1[i] - '0');
                i++;
            }
            //跳过点
            i++;
            long long num2 = 0;
            //从下一个点前截取数字
            while(j < n2 && version2[j] != '.'){
                num2 = num2 * 10 + (version2[j] - '0');
                j++;
            }
            //跳过点
            j++;
            //比较数字大小
            if(num1 > num2)
                return 1;
            if(num1 < num2)
                return -1;
        }
        //版本号相同
        return 0;
    }
