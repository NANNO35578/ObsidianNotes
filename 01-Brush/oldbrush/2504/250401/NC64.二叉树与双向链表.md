[将一个二叉搜索树转换为双向链表](https://www.nowcoder.com/profile/386104436/codeBookDetail?submissionId=456683891)
- 中序遍历即链表顺序 在遍历过程中连接即可
```cpp
class Solution {
   public:
    TreeNode* head = nullptr;
    TreeNode* point = nullptr;
    //? 思路为中序遍历
    void collectNode(TreeNode* root) {
        if (root == nullptr) return;
        point->right = root;
        root->left = point;
        point = root;
    }
    void inorder(TreeNode* root) {
        if (root == nullptr) return;
        inorder(root->left);
        // collect root;
        if (root != head) collectNode(root);
        inorder(root->right);
    }
    TreeNode* Convert(TreeNode* pRootOfTree) {
		if(pRootOfTree==nullptr)return nullptr;
        if (pRootOfTree->left == nullptr&&pRootOfTree->right==nullptr) return pRootOfTree;
        head = pRootOfTree;
        while (head != nullptr&&head->left!=nullptr) {
            if (head->left != nullptr) {
                head = head->left;
            }
        }
                point = head;
		inorder(pRootOfTree);
		return head;
    }
};
```

##### 使用栈存储父节点
```cpp
    TreeNode* treeToDoublyList(TreeNode* root) {
        if (!root) return nullptr;

        std::stack<TreeNode*> stack;
        TreeNode* head = nullptr;
        TreeNode* prev = nullptr;
        TreeNode* current = root;

        while (current || !stack.empty()) {
            while (current) {
                stack.push(current);
                current = current->left;
            }

            current = stack.top();
            stack.pop();

            // 处理当前节点
            if (!head) {
                head = current; // 记录链表的头节点
            }
            if (prev) {
                prev->right = current; // 前一个节点的右指针指向当前节点
                current->left = prev; // 当前节点的左指针指向前一个节点
            }
            prev = current; // 更新前一个节点

            current = current->right; // 继续处理右子树
        }

        // 将头和尾连接
        head->left = prev;
        prev->right = head;

        return head;
    }
```

##### Morris 遍历是一种更高效的遍历方法，它在遍历过程中不使用额外的空间，而是通过修改树的结构来实现。这种方法的时间复杂度仍然是 O(n)，但空间复杂度是 O(1)
```cpp
class Solution {
public:
    TreeNode* treeToDoublyList(TreeNode* root) {
        if (!root) return nullptr;

        TreeNode* head = nullptr;
        TreeNode* prev = nullptr;
        TreeNode* current = root;

        while (current) {
            if (!current->left) {
                // 处理当前节点
                if (!head) head = current; // 记录链表的头节点
                if (prev) {
                    prev->right = current;
                    current->left = prev;
                }
                prev = current;
                current = current->right; // 移动到右子树
            } else {
                // 找到前驱节点
                TreeNode* pre = current->left;
                while (pre->right && pre->right != current) {
                    pre = pre->right;
                }
                if (!pre->right) {
                    pre->right = current; // 创建临时链接
                    current = current->left; // 移动到左子树
                } else {
                    pre->right = nullptr; // 恢复树的结构
                    // 处理当前节点
                    if (!head) head = current;
                    if (prev) {
                        prev->right = current;
                        current->left = prev;
                    }
                    prev = current;
                    current = current->right; // 移动到右子树
                }
            }
        }

        // 将头和尾连接
        head->left = prev;
        prev->right = head;

        return head;
    }
};
```
