```cpp
    void reverWord(string& s, int beg, int ed) {
        if (beg == ed) {
            s[beg] += s[beg] >= 'A' && s[beg] < 'a' ? 32 : -32;
            return;
        }
        while (beg < ed) {
            s[beg] += s[beg] >= 'A' && s[beg] < 'a' ? 32 : -32;
            s[ed] += s[ed] >= 'A' && s[ed] < 'a' ? 32 : -32;
            swap(s[beg], s[ed]);
            ++beg;
            --ed;
            if(beg==ed){
                s[beg] += s[beg] >= 'A' && s[beg] < 'a' ? 32 : -32;
            }
        }
    }
    string trans(string s, int n) {
        // write code here
        // 反转整个字符串
        reverse(s.begin(), s.end());
        // 反转单词 变大小写
        int beg = 0;
        int point;
        for (point = 0; point < n; ++point) {
            if (s[point] == ' ') {
                reverWord(s, beg, point - 1);
                beg = point + 1;
            }
        }
        reverWord(s,beg,point-1);
        return s;
    }
```

##### 使用栈存储单词
- 单个字符入栈1， 遇到空格压入栈2 栈2再压入一个空格 最后将栈2内容倒出来处理
```cpp

void processStack(std::stack<char>& charStack, std::stack<char>& spaceStack) {
    while (!charStack.empty()) {
        char c = charStack.top();
        charStack.pop();
        // 切换大小写
        if (c >= 'A' && c <= 'Z') {
            c += 32; // 转为小写
        } else if (c >= 'a' && c <= 'z') {
            c -= 32; // 转为大写
        }
        spaceStack.push(c);
    }
}

std::string trans(std::string s) {
    std::stack<char> charStack; // 用于存储字符
    std::stack<char> spaceStack; // 用于存储单词和空格

    for (char c : s) {
        if (c != ' ') {
            charStack.push(c); // 入栈1
        } else {
            // 处理栈1中的字符
            processStack(charStack, spaceStack);
            spaceStack.push(' '); // 压入空格
        }
    }
    // 处理最后一个单词
    processStack(charStack, spaceStack);

    // 构建结果字符串
    std::string result;
    while (!spaceStack.empty()) {
        result += spaceStack.top();
        spaceStack.pop();
    }

    return result;
}
```
