[394. 字符串解码](https://leetcode.cn/problems/decode-string/)

- 将字符逐个入栈
- 若为`]` 则要进行出栈操作，取出栈顶元素待编码子串(直到栈顶为`[`) 取出待编码次数
- 编码，放回栈中


```cpp
    string decodeString(string s) {
       stack<char> stk;
       string ans="";
       string subStr="";
       string tempStr="";

       for (char c : s) {
           if (c == ']') {
               while (stk.top() != '[') {
                   subStr += stk.top();
                   stk.pop();
               }
               reverse(subStr.begin(),subStr.end());

               stk.pop();
               int offset = 1;
               int times = 0;
               while (!stk.empty()&&'0' <= stk.top() && '9' >= stk.top()) {
                   times += offset * (stk.top() - '0');
                   offset *= 10;
                   stk.pop();
               }

               while (times) {
                   tempStr += subStr;
                   --times;
               }
               for (char c : tempStr) {
                   stk.push(c);
               }
               tempStr.clear();
               subStr.clear();
           } else {
               stk.push(c);
           }
       }
       while (!stk.empty()) {
           ans += stk.top();
           stk.pop();
       }
       reverse(ans.begin(),ans.end());
       return ans;
    }
```


##### 递归
- 如果当前位置为数字位，那么后面一定包含一个用方括号表示的字符串，即属于这种情况：`k[...]`：
    - 我们可以先解析出一个数字，然后解析到了左括号，递归向下解析后面的内容，遇到对应的右括号就返回，此时我们可以根据解析出的数字 x 解析出的括号里的字符串 s′ 构造出一个新的字符串 `x×s′`；
    - 我们把 `k[...]` 解析结束后，再次调用递归函数，解析右括号右边的内容。
- 如果当前位置是字母位，那么我们直接解析当前这个字母，然后递归向下解析这个字母后面的内容。

```cpp
class Solution {
public:
    string src; 
    size_t ptr;

    int getDigits() {
        int ret = 0;
        while (ptr < src.size() && isdigit(src[ptr])) {
            ret = ret * 10 + src[ptr++] - '0';
        }
        return ret;
    }

    string getString() {
        if (ptr == src.size() || src[ptr] == ']') {
            // String -> EPS
            return "";
        }

        char cur = src[ptr]; int repTime = 1;
        string ret;

        if (isdigit(cur)) {
            // String -> Digits [ String ] String
            // 解析 Digits
            repTime = getDigits(); 
            // 过滤左括号
            ++ptr;
            // 解析 String
            string str = getString(); 
            // 过滤右括号
            ++ptr;
            // 构造字符串
            while (repTime--) ret += str; 
        } else if (isalpha(cur)) {
            // String -> Char String
            // 解析 Char
            ret = string(1, src[ptr++]);
        }
        
        return ret + getString();
    }

    string decodeString(string s) {
        src = s;
        ptr = 0;
        return getString();
    }
};
```
