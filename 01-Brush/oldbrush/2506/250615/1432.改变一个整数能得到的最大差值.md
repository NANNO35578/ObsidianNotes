[1432. 改变一个整数能得到的最大差值](https://leetcode.cn/problems/max-difference-you-can-get-from-changing-an-integer/)


##### 贪心
- 求最大数: 最高位若为9，找到不为9的最高位，设该位为x，将剩余为x的位替换为9 
	- 若最高位不为9，设该位为x，将为x的位替换为9
- 求最小数: 若最高位为1，找到不为1的最高位，设该位为x，将剩余为x的位替换为1
	- 若最高位不为1，设该位为x，将为x的位替换为1

```cpp
class Solution {
public:
    int replaceWithNine(int num) {
        // 找到最高位
        int highestDigit = num;
        int result = 0;
        std::vector<int> bitsNum;

        do {
            bitsNum.push_back(highestDigit % 10);
            highestDigit /= 10;
        } while (highestDigit > 0);

        int h = -1;
        int rp = 9;
        for (int i = bitsNum.size() - 1; i >= 0; --i) {
            if (i + 1 == bitsNum.size()) {
                if (bitsNum[i] != 9) {
                    h = bitsNum[i];
                    bitsNum[i] = 9;
                }
            }

            if (h != -1) {
                if (bitsNum[i] == h) {
                    bitsNum[i] = 9;
                }
            } else if (bitsNum[i] != 9) {
                h = bitsNum[i];
                bitsNum[i] = 9;
            }
        }

        int factor = 1;

        for (int i = 0; i < bitsNum.size(); i++) {
            result += bitsNum[i] * factor;
            factor *= 10; // 更新位权
        }

        return result;
    }

    int modifyNumber(int num) {
        // 找到最高位
        int highestDigit = num;
        int result = 0;
        std::vector<int> bitsNum;

        do {
            bitsNum.push_back(highestDigit % 10);
            highestDigit /= 10;
        } while (highestDigit > 0);

        int h = -1;
        int rp = 1;
        for (int i = bitsNum.size() - 1; i >= 0; --i) {
            if (i + 1 == bitsNum.size()) {
                if (bitsNum[i] != 1) {
                    rp = 1;
                    h = bitsNum[i];
                    bitsNum[i] = 1;
                } else
                    rp = 0;
            }

            if (h != -1) {
                if (bitsNum[i] == h) {
                    bitsNum[i] = rp;
                }
            } else if (bitsNum[i] != 1&&bitsNum[i]!=0) {
                h = bitsNum[i];
                bitsNum[i] = rp;
            }
        }

        int factor = 1;

        for (int i = 0; i < bitsNum.size(); i++) {
            result += bitsNum[i] * factor;
            factor *= 10; // 更新位权
        }

        return result;
    }

    int maxDiff(int num) { return replaceWithNine(num) - modifyNumber(num); }
};
```

##### *优化*
```cpp
class Solution {
public:
    int maxDiff(int num) {
        auto replace = [](string& s, char x, char y) {
            for (char& digit: s) {
                if (digit == x) {
                    digit = y;
                }
            }
        };

        string min_num = to_string(num);
        string max_num = to_string(num);
        // 找到一个高位替换成 9
        for (char digit: max_num) {
            if (digit != '9') {
                replace(max_num, digit, '9');
                break;
            }
        }

        // 将最高位替换成 1
        // 或者找到一个与最高位不相等的高位替换成 0
        for (int i = 0; i < min_num.size(); ++i) {
            char digit = min_num[i];
            if (i == 0) {
                if (digit != '1') {
                    replace(min_num, digit, '1');
                    break;
                }
            }
            else {
                if (digit != '0' && digit != min_num[0]) {
                    replace(min_num, digit, '0');
                    break;
                }
            }
        }

        return stoi(max_num) - stoi(min_num);
    }
};
```

#### 枚举

根据题目的描述，我们可以任意选择两个数字 x 和 y，并将 num 中的所有 x 替换成 y。由于 x 和 y 的取值范围均为 `[0,9]`，那么我们最多只有 10×10=100 种不同的替换方法。

因此我们可以使用两重循环枚举所有的替换方法。在得到的所有新整数中，找出最大以及最小的赋予 a 和 b。

```cpp
class Solution {
public:
    int maxDiff(int num) {
        auto change = [&](int x, int y) {
            string num_s = to_string(num);
            for (char& digit: num_s) {
                if (digit - '0' == x) {
                    digit = '0' + y;
                }
            }
            return num_s;
        };

        int min_num = num;
        int max_num = num;
        for (int x = 0; x < 10; ++x) {
            for (int y = 0; y < 10; ++y) {
                string res = change(x, y);
                // 判断是否有前导零
                if (res[0] != '0') {
                    int res_i = stoi(res);
                    min_num = min(min_num, res_i);
                    max_num = max(max_num, res_i);
                }
            }
        }

        return max_num - min_num;
    }
};
```
