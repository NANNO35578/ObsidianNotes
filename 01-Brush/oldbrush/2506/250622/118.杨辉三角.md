[118. 杨辉三角](https://leetcode.cn/problems/pascals-triangle/)


- 初始化前两行
- 从第三行开始 首先当前行首添加一个1 从下标1开始遍历上一行`preNums[i]`，当前行`curNums[i] = preNums[i-1]+ preNums[i]` 完成遍历在当前行末尾再加个1

```cpp
    vector<vector<int>> generate(int numRows) {
        vector<vector<int>> ans;
        vector<int> lastRow = {1};
        vector<int> pushRow;
        ans.push_back(lastRow);
        if (numRows == 1)
            return ans;

        for (int i = 2; i <= numRows; ++i) {
            pushRow.push_back(1);
            for (int j = 1; j < lastRow.size(); ++j) {
                pushRow.push_back(lastRow[j - 1] + lastRow[j]);
            }
            pushRow.push_back(1);
            ans.push_back(pushRow);
            lastRow=pushRow;
            pushRow.clear();
        }
        return ans;
    }
```

##### 优化
```cpp
    vector<vector<int>> generate(int numRows) {
        vector<vector<int>> ans;
        
        if (numRows <= 0) return ans; // 处理边界情况
        ans.push_back({1}); // 第一行

        for (int i = 1; i < numRows; ++i) {
            vector<int> currentRow(i + 1, 1); // 每行首尾都是1
            const auto& lastRow = ans.back();
            
            for (int j = 1; j < i; ++j) {
                currentRow[j] = lastRow[j - 1] + lastRow[j];
            }
            
            ans.push_back(currentRow);
        }
        
        return ans;
    }
```

    vector<vector<int>> generate(int numRows) {
        vector<vector<int>> ret(numRows);
        for (int i = 0; i < numRows; ++i) {
            ret[i].resize(i + 1);
            ret[i][0] = ret[i][i] = 1;
            for (int j = 1; j < i; ++j) {
                ret[i][j] = ret[i - 1][j] + ret[i - 1][j - 1];
            }
        }
        return ret;
    }
