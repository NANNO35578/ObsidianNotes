
[437. 路径总和 III](https://leetcode.cn/problems/path-sum-iii/)

----

#### 前缀和 + 哈希表

1. 设 `pre[i]` 为从根节点到当前节点的路径和
2. 我们想找以当前节点 `u` 结尾的所有「向下」路径，使得路径和 = `targetSum`，等价于找之前某个前缀和 `pre[j] = pre[i] - targetSum`
3. 用一个哈希表 `cnt` 维护当前 DFS 路径上的各个 `pre[x]` 出现次数。初始时 `cnt[0] = 1`（空前缀）。
4. 每到一个节点，更新当前前缀和 `cur += u->val`，答案增加 `cnt[cur - targetSum]`。
5. 将 `cnt[cur]++` 后继续递归左右子树。回溯时再 `cnt[cur]--`。

```cpp title:"总时间 O(n)，空间 O(n)"
class Solution {
public:
    int pathSum(TreeNode* root, int targetSum) {
        this->target = targetSum;
        ans = 0;
        cnt.clear();
        cnt[0] = 1; // 空前缀
        dfs(root, 0);
        return ans;
    }
private:
    unordered_map<long long,int> cnt;
    int target;
    int ans;

    void dfs(TreeNode* u, long long cur) {
        if (!u) return;
        cur += u->val;
        // 以 u 为「结尾」的路径中，和为 target 的条数
        if (cnt.count(cur - target))
            ans += cnt[cur - target];

        // 加入当前前缀，进入子树
        cnt[cur]++;
        dfs(u->left, cur);
        dfs(u->right, cur);
        // 回溯
        cnt[cur]--;
    }
};
```

#### 离散化数组

把所有前缀和先离散化到 `[1..M]`，再用定长数组 `cnt[M+1]` 做「前缀和频次表」，更新/查询都 O(1)，加上离散化的 $O(n log_(n))$ 预处理，总体 $O(n log_(n))$

1. **第一遍 DFS** — 计算从根到每个节点的前缀和 `S[u]`，并把它们都存到一个 `vector<long long> all` 里，同时把 `0` 也加进去。
2. 对 `all` 去重、排序，建立一个 `map<long long,int> id`，把每个前缀和映射到 `[1…M]` 的小整数。
3. 用一个大小为 `M+1` 的 `vector<int> cnt(M+1)` 替代哈希表，支持 O(1) 的 `cnt[k]++/--` 和 `cnt[k]` 读取。
4. **第二遍 DFS** — 同方法一，但把 `unordered_map` 的部分改成对离散化后的数组 `cnt` 操作。

```cpp
class Solution {
public:
    int pathSum(TreeNode* root, int targetSum) {
        target = targetSum;
        // 1. 收集所有前缀和
        all.clear();
        all.push_back(0);    // 初始前缀和
        dfs_collect(root, 0);
        // 2. 离散化
        sort(all.begin(), all.end());
        all.erase(unique(all.begin(), all.end()), all.end());
        // 3. 建 map
        M = all.size();
        // 4. 第二次 DFS 维护离散化数组
        cnt.assign(M+1, 0);
        ans = 0;
        cur = 0;
        cnt[getId(0)] = 1;   // 空前缀出现 1 次
        dfs_count(root);
        return ans;
    }

private:
    vector<long long> all;
    int M;
    int target, ans;
    long long cur;
    vector<int> cnt;

    // 第一次 DFS：收集前缀和
    void dfs_collect(TreeNode* u, long long sum) {
        if (!u) return;
        sum += u->val;
        all.push_back(sum);
        dfs_collect(u->left,  sum);
        dfs_collect(u->right, sum);
    }

    // 把一个前缀和映射到离散化下标 [1..M]
    int getId(long long x) {
        // 二分查找
        auto it = lower_bound(all.begin(), all.end(), x);
        return int(it - all.begin()) + 1;
    }

    // 第二次 DFS：实际统计
    void dfs_count(TreeNode* u) {
        if (!u) return;
        cur += u->val;
        // 查 cnt[cur - target]
        long long need = cur - target;
        auto it = lower_bound(all.begin(), all.end(), need);
        if (it != all.end() && *it == need) {
            ans += cnt[int(it - all.begin()) + 1];
        }
        // 加入 cur
        int id_cur = getId(cur);
        cnt[id_cur]++;
        // 递归
        dfs_count(u->left);
        dfs_count(u->right);
        // 回溯
        cnt[id_cur]--;
        cur -= u->val;
    }
};
```
