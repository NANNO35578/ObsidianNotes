[1931. 用三种不同颜色为网格涂色](https://leetcode.cn/problems/painting-a-grid-with-three-different-colors/)

- 将网格按列来 DP，每一列是一个长度为 m 的序列，每个位置有 3 种颜色，要求同列相邻两格颜色不同，所以合法状态数量为 S=3⋅2^(m−1)
- 把每个合法状态用一个 0..3^m−1 之间的整数（3 进制）来编码，并预先筛掉有相邻相同颜色的状态，得到合法状态列表 `states[]`
- 预先计算出任意两个合法状态之间能否左右相邻（即对应行的颜色都不同），构造转移表 `can[i][j]`
- 用 `DP[i][j]` 表示涂到第 i 列时，第 i 列取 `states[j]` 的方案数。转移：  


$$DP[i][j] = ∑_{k: can[k][j]} DP[i−1][k]$$


- 初始 `DP[0][*]` 全置 1，答案为 `∑_j DP[n−1][j] % mod`。  
- 如果 m>n，则先交换 m,n，以保证状态空间最小  


<dl>
  <dt></dt>
    <dd>- 把「列」当作一步，状态是当前列的 m 个格子涂色方案（不含同列相邻相同）</dd>
    <dd>- 相邻两列之间要求同一行颜色不同</dd>
    <dd>- DP 按列逐步累加方案数，最终取模即可</dd>
</dl>

```cpp
static const int MOD = 1000000007;

// 快速加法取模
inline void addmod(int &a, int b) {
    a += b;
    if (a >= MOD) a -= MOD;
}

int colorTheGrid(int m, int n) {
    // 保证 m <= n，以减少状态数
    if (m > n) swap(m, n);

    // 生成所有长度为 m 的 3 进制数，筛掉同列相邻相同的
    int maxState = 1;
    for (int i = 0; i < m; i++) maxState *= 3;
    vector<int> states;
    states.reserve(maxState);

    auto valid = [&](int s) {
        int prev = -1;
        for (int i = 0; i < m; i++) {
            int c = s % 3;
            s /= 3;
            if (c == prev) return false;
            prev = c;
        }
        return true;
    };

    for (int s = 0; s < maxState; s++) {
        if (valid(s)) states.push_back(s);
    }

    int S = states.size();
    // 预计算任意两状态是否能左右相邻（同一行颜色不同）
    vector<vector<int>> can(S, vector<int>(S, 0));
    for (int i = 0; i < S; i++) {
        for (int j = 0; j < S; j++) {
            int a = states[i], b = states[j];
            bool ok = true;
            for (int k = 0; k < m; k++) {
                if (a % 3 == b % 3) {
                    ok = false;
                    break;
                }
                a /= 3; b /= 3;
            }
            can[i][j] = ok;
        }
    }

    // 滚动数组做 DP
    vector<int> dp(S, 1), ndp(S, 0);

    // 列索引从第 2 列到第 n 列做转移
    for (int col = 1; col < n; col++) {
        fill(ndp.begin(), ndp.end(), 0);
        for (int j = 0; j < S; j++) if (dp[j]) {
            int v = dp[j];
            for (int k = 0; k < S; k++) {
                if (can[j][k]) {
                    addmod(ndp[k], v);
                }
            }
        }
        dp.swap(ndp);
    }

    // 所有列完成后，把最后一列所有状态求和
    int ans = 0;
    for (int x : dp) addmod(ans, x);
    return ans;
}
```

##### 矩阵快速幂

除了上面按列 $O(n\cdot S^2)$ 滚动 DP（其中 S 是「合法列状态」的数量）外，还可以把转移看作一个 $S\times S$的邻接矩阵，利用矩阵快速幂把时间复杂度降到 $O(S^3\log n)$

 假设 $m\le n$，若不满足可先 swap：

1. 枚举出所有「合法列状态」states，令其数目为 $S$。
2. 构造 $S\times S$的矩阵 $M$，令
	$$M[i][j]= \begin{cases} 1,&\text{如果 states[i] 与 states[j] 可左右相邻（对应行颜色互不相同）},\\ 0,&\text{否则}. \end{cases}$$
    
3. 初始时第 $1$ 列可任意取一个状态，方案为向量 $v\in\mathbb{Z}^S$ 且 $v_i=1\ \forall i$
4. 则第 $n$ 列的方案数向量为
    $v^{(n)} \;=\; M^{\,n-1}\;v$
5. 答案就是 $v^{(n)}$ 所有分量之和，再对 $10^9+7$ 取模。


```cpp
#include <bits/stdc++.h>
using namespace std;
static const int MOD = 1000000007;
using ll = long long;

// 模矩阵乘法：C = A * B
vector<vector<int>> matMul(const vector<vector<int>>& A, const vector<vector<int>>& B) {
    int S = A.size();
    vector<vector<int>> C(S, vector<int>(S, 0));
    for (int i = 0; i < S; i++) {
        for (int k = 0; k < S; k++) if (A[i][k]) {
            ll aik = A[i][k];
            for (int j = 0; j < S; j++) {
                C[i][j] = (C[i][j] + aik * B[k][j]) % MOD;
            }
        }
    }
    return C;
}

// 矩阵快速幂：M^e
vector<vector<int>> matPow(vector<vector<int>> M, long long e) {
    int S = M.size();
    vector<vector<int>> R(S, vector<int>(S, 0));
    for (int i = 0; i < S; i++) R[i][i] = 1;
    while (e > 0) {
        if (e & 1) R = matMul(R, M);
        M = matMul(M, M);
        e >>= 1;
    }
    return R;
}

int colorTheGrid(int m, int n) {
    if (m > n) swap(m, n);
    int maxState = 1;
    for (int i = 0; i < m; i++) maxState *= 3;

    // 1) 枚举合法的列状态
    vector<int> states;
    auto valid = [&](int s) {
        int prev = -1;
        for (int i = 0; i < m; i++) {
            int c = s % 3; s /= 3;
            if (c == prev) return false;
            prev = c;
        }
        return true;
    };
    for (int s = 0; s < maxState; s++)
        if (valid(s)) states.push_back(s);
    int S = states.size();

    // 2) 构造转移矩阵 M
    vector<vector<int>> M(S, vector<int>(S, 0));
    for (int i = 0; i < S; i++) {
        for (int j = 0; j < S; j++) {
            int a = states[i], b = states[j];
            bool ok = true;
            for (int k = 0; k < m; k++) {
                if (a % 3 == b % 3) { ok = false; break; }
                a /= 3; b /= 3;
            }
            if (ok) M[i][j] = 1;
        }
    }

    // 3) 计算 M^(n-1)
    auto Mexp = matPow(M, n - 1);

    // 4) 初始向量 v 全 1，计算 v' = Mexp * v
    //    最终答案 = sum(v'[i]) % MOD
    ll ans = 0;
    for (int i = 0; i < S; i++) {
        // v'[i] = sum_j Mexp[i][j] * 1
        for (int j = 0; j < S; j++) {
            ans = (ans + Mexp[i][j]) % MOD;
        }
    }
    return ans;
}
```
