[994. 腐烂的橘子](https://leetcode.cn/problems/rotting-oranges/)

1. **初始化数据结构**：
    - 使用一个队列来存储腐烂橘子的坐标。
    - 使用一个变量来记录新鲜橘子的数量。
    - 记录时间（分钟数）作为 BFS 的层数。
2. **遍历网格**： 遍历整个网格，找到所有腐烂橘子的位置并将其放入队列，同时统计新鲜橘子的数量。
3. **BFS 过程**：
    - 从队列中取出腐烂的橘子，并检查其四个方向（上、下、左、右）。
    - 对于每个方向，如果相邻的单元格是新鲜橘子（值为 1），则将其腐烂（将其值改为 2）并加入队列，同时减少新鲜橘子的计数。
4. **时间计数**： 每次处理完一层的腐烂橘子后，时间加 1，直到队列为空。
5. **判断结果**： 如果新鲜橘子的数量为 0，返回时间；如果队列为空但仍有新鲜橘子，返回 -1。

```cpp
int orangesRotting(vector<vector<int>>& grid) {
    int rows = grid.size();
    int cols = grid[0].size();
    queue<pair<int, int>> q;
    int freshCount = 0;

    // 初始化队列和统计新鲜橘子数量
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            if (grid[i][j] == 2) {
                q.push({i, j});
            } else if (grid[i][j] == 1) {
                freshCount++;
            }
        }
    }

    int minutes = 0;
    vector<vector<int>> directions = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};

    // BFS
    while (!q.empty()) {
        int size = q.size();
        for (int i = 0; i < size; i++) {
            auto [x, y] = q.front();
            q.pop();
            for (const auto& dir : directions) {
                int newX = x + dir[0];
                int newY = y + dir[1];
                // 检查边界和新鲜橘子
                if (newX >= 0 && newX < rows && newY >= 0 && newY < cols && grid[newX][newY] == 1) {
                    grid[newX][newY] = 2; // 腐烂
                    q.push({newX, newY});
                    freshCount--; //减少新鲜橘子的数量
                }
            }
        }
        if (!q.empty()) {
            minutes++; // 只有在队列不为空时才增加时间
        }
    }

    return freshCount == 0 ? minutes : -1; // 如果没有新鲜橘子，返回时间，否则返回 -1
}
```
