[统计是给定字符串前缀的数目](https://leetcode.cn/problems/count-prefixes-of-a-given-string/description/)
- 无特殊要求
```cpp
class Solution {    //! string可以自己比较 😒😒
public:
    int countPrefixes(vector<string>& words, string s) {
        int ans = words.size();
        for (string& sub : words)
           for (int i = 0; i < sub.size(); ++i)
               if (s[i] != sub[i]) {
                   --ans;
                   break;
               }
       return ans;
    }
};
```
