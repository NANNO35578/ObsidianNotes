    void travelF(TreeNode* root, vector<int>* nums) {
        if (root == nullptr) return;  // 空返回
        nums->push_back(root->val);
        travelF(root->left, nums);
        travelF(root->right, nums);
    }
    vector<int> preorderTraversal(TreeNode* root) {
        // write code here
        //* 递归写法

        //* 循环写法
        if(root==nullptr)return vector<int>({});
        vector<int> ans;
        stack<TreeNode*> parents;
        parents.push(root);
        while (!parents.empty()) {
            TreeNode* top = parents.top();
            ans.push_back(top->val);
            parents.pop();
            if (top->right != nullptr) {
                parents.push(top->right);
            }
            if (top->left != nullptr) parents.push(top->left);
        }
        return ans;
    }
