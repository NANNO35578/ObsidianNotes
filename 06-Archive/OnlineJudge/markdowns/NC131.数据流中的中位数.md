[数据流中的中位数](https://www.nowcoder.com/profile/386104436/codeBookDetail?submissionId=457592048) 
- 考虑到会有数据插入 同时需要快速定位到数据从小到大的中位数
- 使用链表存储有序数据
```cpp

class Solution {
private:
    struct ListNode {
        int val;
        struct ListNode* next;
        ListNode(int x) : val(x), next(nullptr) {}
    };
    int listSize = 0;
    ListNode* head = nullptr;
    ListNode* End = nullptr;
    int maxI;
    int minI;

public:
    void Insert(int num) {
        ++listSize;
        if (head == nullptr) {
            head = new ListNode(num);
            End = head;
            maxI = num;
            minI = num;
            return;
        }
        if (num <= minI) {
            ListNode* newHead = new ListNode(num);
            newHead->next = head;
            head = newHead;
            minI = num;
            return;
        } else if (num >= maxI) {
            ListNode* newEnd = new ListNode(num);
            End->next = newEnd;
            End = newEnd;
            maxI = num; // Update maxI
            return;
        }
        
        ListNode* newNode = new ListNode(num);
        ListNode* point = head;
        while (point->next != nullptr) {
            if (num < point->next->val) {
                newNode->next = point->next;
                point->next = newNode;
                return;
            }
            point = point->next;
        }
        // In case we reach the end of the list and need to insert
        point->next = newNode;
    }

    double GetMedian() {
        if (listSize == 1) return (double)head->val;
        if (listSize == 2) return (double)(head->val + End->val) / 2;

        ListNode* point = head;
        int mid = listSize / 2;

        for (int i = 0; i < mid - 1; ++i) {
            point = point->next;
        }
        
        if (listSize % 2 == 0) {
            return (point->val + point->next->val) / 2.0;
        } else {
            return point->next->val; // For odd, return the middle element
        }
    }
};
```
