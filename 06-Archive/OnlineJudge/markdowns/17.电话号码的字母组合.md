[17. 电话号码的字母组合](https://leetcode.cn/problems/letter-combinations-of-a-phone-number/)

1. **映射数组**：用一个数组来存储数字到字母的映射。
2. **回溯函数**：`backtrack` 函数用于生成字母组合。
    - 当当前组合的长度等于输入的长度时，将其添加到结果中。
    - 否则，获取当前数字对应的字母，循环添加每个字母并递归调用。


```cpp
    void backtrack(vector<string>& result, string& current,
                   const string& digits, int index,
                   const vector<string>& mappings) {
        if (index == digits.size()) {  // 如果当前组合长度等于输入长度
            result.push_back(current); // 将当前组合添加到结果中
            return;
        }

        int digit = digits[index] - '0';         // 当前数字
        const string& letters = mappings[digit]; // 获取对应字母

        for (char letter : letters) {
            current.push_back(letter); // 添加当前字母
            backtrack(result, current, digits, index + 1,
                      mappings); // 递归下一层
            current.pop_back();  // 回溯，移除最后一个字母
        }
    }

    vector<string> letterCombinations(string digits) {
        if (digits.empty())
            return {}; // 如果输入为空，返回空向量

        // 数字到字母的映射
        vector<string> mappings = {"",    "",    "abc",  "def", "ghi",
                                   "jkl", "mno", "pqrs", "tuv", "wxyz"};

        vector<string> result;
        string current;
        backtrack(result, current, digits, 0, mappings);
        return result;
    }
```
