[74. 搜索二维矩阵](https://leetcode.cn/problems/search-a-2d-matrix/)

- 二分查找
- 对第一列进行二分查找，确定可能存在的两行
- 对两行进行二分查找

```cpp
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int t = 0;
        int d = matrix.size() - 1;
        int m;
        while (t < d) {
            m = (t + d) / 2;
            if (matrix[m][0] == target) {
                return true;
            } else if (matrix[m][0] < target) {
                if (t + 1 == d )
                    break;
                t = m;
            } else {
                d=m;
            }
        }
        if(t+1==d){
            auto it = lower_bound(matrix[t].begin(), matrix[t].end(), target);
            if (it != matrix[t].end() && *it == target) {
                return true;
            }
        }

            auto it = lower_bound(matrix[d].begin(), matrix[d].end(), target);
            if (it != matrix[d].end() && *it == target) {
                return true;
            }

        return false;
    }
```

##### 优化
```cpp
    bool searchMatrix(vector<vector<int>> matrix, int target) {
        auto row = upper_bound(matrix.begin(), matrix.end(), target, [](const int b, const vector<int> &a) {
            return b < a[0];
        });
        if (row == matrix.begin()) {
            return false;
        }
        --row;
        return binary_search(row->begin(), row->end(), target);
    }
```


##### 一次二分查找
- 若将矩阵每一行拼接在上一行的末尾，则会得到一个升序数组，我们可以在该数组上二分找到目标元素。
- 代码实现时，可以二分升序数组的下标，将其映射到原矩阵的行和列上。

```cpp
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int m = matrix.size(), n = matrix[0].size();
        int low = 0, high = m * n - 1;
        while (low <= high) {
            int mid = (high - low) / 2 + low;
            int x = matrix[mid / n][mid % n];
            if (x < target) {
                low = mid + 1;
            } else if (x > target) {
                high = mid - 1;
            } else {
                return true;
            }
        }
        return false;
    }
```
