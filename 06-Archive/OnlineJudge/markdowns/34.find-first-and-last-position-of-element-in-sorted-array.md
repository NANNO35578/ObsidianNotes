[排序数组查找第一次和最后一次出现位置](https://leetcode.cn/problems/find-first-and-last-position-of-element-in-sorted-array/description/)
- 这道题可以看作是自己实现 C++ 里的 `lower_bound` 和 `upper_bound` 函数 同样使用二分法
```cpp
    int lower(vector<int>& nums, int target) {
        int l = 0, r = nums.size(), mid;  

        while (l < r) {
            mid = l + (r - l) / 2;
            if (nums[mid] >= target)
                r = mid;
            else
                l = mid + 1;
        }
        return l;
    }
    int upper(vector<int>& nums, int target) {
        int l = 0, r = nums.size(), mid;
  
        while (l < r) {
            mid = l + (r - l) / 2;
            if (nums[mid] > target)
                            r = mid;
            else
                l = mid + 1;
        }
        return l;
    }
  
    vector<int> searchRange(vector<int>& nums, int target) {
        if (nums.empty()) return vector<int>{-1, -1};
        int lowerI = lower(nums, target);
        int upperI = upper(nums, target) - 1;
        if (lowerI == nums.size() || nums[lowerI] != target)
            return vector{-1, -1};
        return vector<int>{lowerI, upperI};
    }
 ```
