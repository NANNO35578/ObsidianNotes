[寻找峰值](https://www.nowcoder.com/profile/386104436/codeBookDetail?submissionId=456678790)
- 利用二分法 若mid点值不满足题解 则若左侧大 峰值可在左 反之可在右
```cpp
    int findPeakElement(vector<int>& nums) {
        // write code here
        if (nums.size() == 1) return 0;
        

        int l = 0;
        int r = nums.size() - 1;
        int mid;
        while (l < r) {
            mid = l + (r - l) / 2;
            if (nums[mid - 1] < nums[mid] && nums[mid] > nums[mid + 1])
                return mid;
            else if (nums[mid] < nums[mid - 1]) {
                // 峰值可在左
                r = mid ;
            } else {
                l = mid + 1;
            }
        }
        return r;
    }
```
