[2411. 按位或最大的最小子数组长度](https://leetcode.cn/problems/smallest-subarrays-with-maximum-bitwise-or/) 


---

## 思路一：逆序维护每一位的「最近出现位置」

对于起点 i，要想子数组 `nums[i..j]` 的 OR 达到最大值（即等于后缀 OR），必须将所有在 `nums[i..n−1]` 中出现的二进制位都「收集」到子数组里。我们可以：

1. 从后往前遍历 i （`i = n−1 … 0`），维护长度为 30（或 31） 的数组 `nextPos[b]`，表示二进制位 b 在当前位置 i 及以后的「最近出现」下标
2. 当遍历到 i 时，先把 `nums[i]` 中出现的那些二进制位的 `nextPos[b]` 更新为 i
3. 那么从 i 出发，想要收集所有后缀能得到的位，就要至少延伸到所有 `nextPos[b]` 的最远值 `farthest`。答案就是 `farthest − i + 1`

时间复杂度 O(n·B)，B=30

```cpp
    vector<int> smallestSubarrays(vector<int>& nums) {
        int n = nums.size();
        vector<int> ans(n);
        const int INF = 1e9;
        // nextPos[b] = 在当前位置及之后，二进制位 b 最近一次出现的位置
        vector<int> nextPos(30, INF);

        // 从后往前
        for (int i = n - 1; i >= 0; --i) {
            // 更新 nums[i] 中各位的 nextPos
            for (int b = 0; b < 30; ++b) {
                if (nums[i] & (1 << b)) {
                    nextPos[b] = i;
                }
            }
            // 计算最远要扩展到的位置
            int farthest = i;
            for (int b = 0; b < 30; ++b) {
                if (nextPos[b] != INF) {
                    farthest = max(farthest, nextPos[b]);
                }
            }
            ans[i] = farthest - i + 1;
        }
        return ans;
    }
```

# 思路二：后缀 OR + 稀疏表（二分查最短右端点）

1. 先预处理出后缀 OR 数组 `sufOR[i] = nums[i] | nums[i+1] | … | nums[n−1]`
2. 构建 OR #幺半群 的稀疏表（Sparse Table），支持 O(1) 区间 OR 查询（均摊 O(B)）
3. 对每个起点 i，用二分在区间 $[i, n−1]$ 中找最小的 j，使得 `OR(i, j) == sufOR[i]`，则 `ans[i] = j − i + 1`

总复杂度 O(n·log n·B)

```cpp
#include <bits/stdc++.h>
using namespace std;

class Solution {
    vector<vector<int>> st;     // st[k][i] 存 2^k 长度区间 OR 的结果
    vector<int> lg2;            // 预处理对数
public:
    // 构建稀疏表
    void buildSparseTable(const vector<int>& nums) {
        int n = nums.size();
        lg2.assign(n + 1, 0);
        for (int i = 2; i <= n; ++i)
            lg2[i] = lg2[i >> 1] + 1;

        int K = lg2[n] + 1;
        st.assign(K, vector<int>(n));
        // k = 0
        for (int i = 0; i < n; ++i)
            st[0][i] = nums[i];
        // k > 0
        for (int k = 1; k < K; ++k) {
            int len = 1 << k;
            for (int i = 0; i + len <= n; ++i) {
                st[k][i] = st[k-1][i] | st[k-1][i + (len >> 1)];
            }
        }
    }

    // 查询区间 OR，O(B) 均摊
    int queryOr(int l, int r) {
        int res = 0;
        int len = r - l + 1;
        for (int k = lg2[len]; k >= 0; --k) {
            int span = 1 << k;
            if (span <= len) {
                res |= st[k][l];
                l += span;
                len -= span;
            }
        }
        return res;
    }

    vector<int> smallestSubarrays(vector<int>& nums) {
        int n = nums.size();
        vector<int> ans(n), sufOR(n);
        // 1. 后缀 OR
        sufOR[n-1] = nums[n-1];
        for (int i = n-2; i >= 0; --i)
            sufOR[i] = sufOR[i+1] | nums[i];

        // 2. 构建稀疏表
        buildSparseTable(nums);

        // 3. 对每个 i 二分查最小 j
        for (int i = 0; i < n; ++i) {
            int target = sufOR[i];
            int l = i, r = n - 1, best = n - 1;
            while (l <= r) {
                int mid = (l + r) >> 1;
                if (queryOr(i, mid) == target) {
                    best = mid;
                    r = mid - 1;
                } else {
                    l = mid + 1;
                }
            }
            ans[i] = best - i + 1;
        }
        return ans;
    }
};
```
