```cpp
    // 快速选择算法
    int quickSelect(vector<int>& nums, int left, int right, int k) {
        if (left == right) {
            return nums[left];
        }

        // 选择一个随机的基准
        int pivotIndex = left + (right - left) / 2;
        int pivotValue = nums[pivotIndex];

        // 将基准移到末尾
        swap(nums[pivotIndex], nums[right]);
        int storeIndex = left;

        // 分区操作
        for (int i = left; i < right; i++) {
            if (nums[i] > pivotValue) {  // 找第 k 大，所以这里是 >
                swap(nums[i], nums[storeIndex]);
                storeIndex++;
            }
        }
        // 把基准移到正确的位置
        swap(nums[storeIndex], nums[right]);

        // 判断基准位置
        if (storeIndex == k) {
            return nums[storeIndex];
        } else if (storeIndex < k) {
            return quickSelect(nums, storeIndex + 1, right, k);
        } else {
            return quickSelect(nums, left, storeIndex - 1, k);
        }
    }
    int findKth(vector<int>& a, int n, int K) {
        // write code here
        // sort(a.begin(),a.end());
        // return a[a.size() - K];
        return quickSelect(a, 0, n - 1, K-1);
    }
```
