    ListNode* Merge(ListNode* pHead1, ListNode* pHead2) {
        // write code here
        if (pHead1 == nullptr) return pHead2;
        if (pHead2 == nullptr) return pHead1;
        ListNode* ans ;
        if (pHead1->val > pHead2->val) {
            ans = pHead2;
            pHead2 = pHead2->next;
        } else {
            ans = pHead1;
            pHead1 = pHead1->next;
        }
 
        ListNode* point = ans;
        while (pHead1 != nullptr && pHead2 != nullptr) {
            if (pHead1->val >= pHead2->val) {
                point->next = pHead2;
                pHead2 = pHead2->next;
            } else {
                point->next = pHead1;
                pHead1 = pHead1->next;
            }
            point = point->next;
        }
        if (pHead2 != nullptr) {
            point->next = pHead2;
        }
        if (pHead1 != nullptr) {
            point->next = pHead1;
        }
        return ans;
    }
