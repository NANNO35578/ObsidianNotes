[合并两个有序数组](https://leetcode.cn/problems/merge-sorted-array/description/)
- 考虑到所有元素应当放入nums1 可以从尾开始使用双指针依次遍历nums1以及nums2 将较大的放入nums1末尾 即可实现
- 注意到nums2中元素可能比nums1元素小 应当在nums1复制完后再复制剩下的nums2
```cpp
/*
 * @lc app=leetcode.cn id=88 lang=cpp
 *
 * [88] 合并两个有序数组
 */
#include <vector>
using namespace std;  

// @lc code=start
class Solution {   
public:
    void merge(vector<int>& nums1, int m, vector<int>& nums2, int n) {
        if (m == 0) {
            nums1 = nums2;
            return;
        }
        if (n == 0) return;
        int pos = m + n - 1;  

        int pm = m - 1;
        int pn = n - 1;
        while (pm >= 0 && pn >= 0) {
            if (nums1[pm] >= nums2[pn]) {
                nums1[pos--] = nums1[pm];
                --pm;
            } else {
                nums1[pos--] = nums2[pn];
                --pn;
            }
        }
        while (pn >= 0) nums1[pos--] = nums2[pn--];
    }
};
```


##### 方法二  直接合并后排序
```cpp
class Solution {
public:
    void merge(vector<int>& nums1, int m, vector<int>& nums2, int n) {
        for (int i = 0; i != n; ++i) {
            nums1[m + i] = nums2[i];
        }
        sort(nums1.begin(), nums1.end());
    }
};
```
