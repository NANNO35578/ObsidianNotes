[链表内指定区间反转](http://nowcoder.com/profile/386104436/codeBookDetail?submissionId=456268869)
 - 定义一函数用于反转指定首尾
 - 添加dummy节点处理m=1时
```cpp
    ListNode* reversePart(ListNode* begin, ListNode* end) {
        ListNode*  endNext=end->next;
        ListNode* prev = nullptr;
        ListNode* curr = begin;
        ListNode* nextNode;
        // Reverse the nodes until we reach the end
        while (curr != end) {
            nextNode = curr->next; // Store the next node
            curr->next = prev;     // Reverse the current node's pointer
            prev = curr;           // Move prev and curr one step forward
            curr = nextNode;
        }
        curr->next = prev;
        return endNext; // Return the new head of the reversed segment
    }
  

    ListNode* reverseBetween(ListNode* head, int m, int n) {
        if (m == n) return head;
  

        ListNode dummy(0); // Create a dummy node to simplify edge cases
        dummy.next = head;
        ListNode* front = &dummy;
  
        // Move front to the node right before the segment to reverse
        for (int i = 1; i < m; ++i) {
            front = front->next;
        }
  
        ListNode* start = front->next; // First node to reverse
        ListNode* end = start;          // Last node to reverse
  

        for (int i = m; i < n; ++i) {
            end = end->next;
        }
  
        // Reverse the part and connect with the rest of the list
        front->next = end;
        start->next = reversePart(start, end);
  
        return dummy.next; // Return the new head of the list
    }
```
