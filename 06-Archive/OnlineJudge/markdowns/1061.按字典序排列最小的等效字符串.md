[1061. 按字典序排列最小的等效字符串](https://leetcode.cn/problems/lexicographically-smallest-equivalent-string/)


1. **初始化并查集**： 创建一个并查集，代表 26 个小写字母。
2. **合并等价字符**： 对于每一对 `s1[i]` 和 `s2[i]`，将它们合并到同一个集合中。
3. **查找最小字符**： 对于 `baseStr` 中的每个字符，找到它的根字符，然后将其替换为对应的最小字符。
4. **构建结果字符串**： 将所有替换后的字符组合成最终的结果字符串。

```cpp
class UnionFind {
public:
    UnionFind(int size) : parent(size), rank(size, 1) {
        for (int i = 0; i < size; ++i) {
            parent[i] = i;
        }
    }

    int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]);
        }
        return parent[x];
    }

    void unionSets(int x, int y) {
        int rootX = find(x);
        int rootY = find(y);
        if (rootX != rootY) {
            if (rank[rootX] > rank[rootY]) {
                parent[rootY] = rootX;
            } else if (rank[rootX] < rank[rootY]) {
                parent[rootX] = rootY;
            } else {
                parent[rootY] = rootX;
                rank[rootX]++;
            }
        }
    }

private:
    std::vector<int> parent;
    std::vector<int> rank;
};

std::string smallestEquivalentString(std::string s1, std::string s2, std::string baseStr) {
    UnionFind uf(26);
    
    for (size_t i = 0; i < s1.length(); ++i) {
        uf.unionSets(s1[i] - 'a', s2[i] - 'a');
    }

    std::vector<char> minChar(26);
    for (char c = 'a'; c <= 'z'; ++c) {
        int root = uf.find(c - 'a');
        minChar[root] = std::min(minChar[root] ? minChar[root] : c, c);
    }

    std::string result;
    for (char c : baseStr) {
        result += minChar[uf.find(c - 'a')];
    }
    
    return result;
}
```

- **并查集**：`UnionFind` 类用于管理字符的并集与查找。
- **合并字符**：通过 `unionSets` 方法将 `s1` 和 `s2` 中的字符合并。
- **查找最小字符**：遍历字符，找到每个字符的根，更新为最小字符。
- **构建结果**：根据 `baseStr` 中的字符构建最终的字符串。

##### 复杂度

- 时间复杂度：O(n + m * α(n))，其中 n 是 `s1` 和 `s2` 的长度，m 是 `baseStr` 的长度，α 是阿克曼函数的反函数，几乎可以认为是常数。
- 空间复杂度：O(1)，因为只使用了固定大小的数组来存储字符的并查集和最小字符。

# 官解
等价字符具有自反性、对称性和传递性，因此可以将整个等价字符集看作无向图中的连通块，在同一个连通块中的点（即字符）相互等价，可以相互替换。我们要为 `baseStr` 中的每个字符找到它所在连通块中字典序最小的字符。

根据题目给定的 s1​ 和 s2​，依次遍历每对字符 `s1​[i]` 和 `s2​[i]`，并将这两个字符所在的连通块进行合并，最后可以得到完整的若干个连通块，就可以进行字符替换了。

维护连通块的合并可以使用「[并查集](https://leetcode.cn/link/?target=https%3A%2F%2Foi-wiki.org%2Fds%2Fdsu%2F)」，为了能够快速找到每个连通块中字典序最小的字符，我们在合并时可以将字典序更小的字符作为代表字符。

**代码**
```cpp
class UnionFind {
    vector<int> f;
    int n;
public:
    UnionFind(int n) : n(n) {
        f.resize(n);
        for (int i = 0; i < n; i++) {
            f[i] = i;
        }
    }
    int find(int x) {
        if (f[x] != x) {
            f[x] = find(f[x]);
        }
        return f[x];
    }
    void unite(int x, int y) {
        x = find(x);
        y = find(y);
        if (x == y) {
            return;
        }
        if(x > y) {
            swap(x, y);
        }
        // 总是让字典序更小的作为集合代表字符
        f[y] = x;
    }
};

class Solution {
public:
    string smallestEquivalentString(string s1, string s2, string baseStr) {
        UnionFind uf(26);
        for(int i = 0; i < s1.size(); i++) {
            int x = s1[i] - 'a';
            int y = s2[i] - 'a';
            uf.unite(x, y);
        }
        for(int i = 0; i < baseStr.size(); i++) {
            baseStr[i] = 'a' + uf.find(baseStr[i] - 'a');
        }
        return baseStr;
    }
};
```

#并查集（Union-Find）
用于处理一些不交集的合并及查询问题。它支持两种主要操作：

1. **查找（Find）**：确定一个元素属于哪个子集。通过路径压缩优化，可以使后续查询更快。
2. **合并（Union）**：将两个子集合并成一个集合。通过按秩合并（Union by rank）优化，可以保持树的高度较小。
##### 应用场景
- 网络连接问题：判断两个节点是否在同一网络中。
- 动态连通性：实时处理连接和查询操作。
- 集合合并：例如，社交网络中的好友关系。
##### 时间复杂度

在使用路径压缩和按秩合并的情况下，两个操作的平均时间复杂度接近于常数时间，具体为 $O(\alpha(n))$，其中 α 是阿克曼函数的反函数，增长极其缓慢。


```cpp
#include <iostream>
#include <vector>

class UnionFind {
public:
    UnionFind(int size) {
        parent.resize(size);
        rank.resize(size, 0);
        for (int i = 0; i < size; i++) {
            parent[i] = i; // 初始化每个节点的父节点为自己
        }
    }

    int find(int p) {
        if (parent[p] != p) {
            parent[p] = find(parent[p]); // 路径压缩
        }
        return parent[p];
    }

    void unionSets(int p, int q) {
        int rootP = find(p);
        int rootQ = find(q);
        if (rootP != rootQ) {
            // 按秩合并
            if (rank[rootP] > rank[rootQ]) {
                parent[rootQ] = rootP;
            } else if (rank[rootP] < rank[rootQ]) {
                parent[rootP] = rootQ;
            } else {
                parent[rootQ] = rootP;
                rank[rootP]++;
            }
        }
    }

private:
    std::vector<int> parent; // 存储每个节点的父节点
    std::vector<int> rank;   // 存储树的高度
};

int main() {
    UnionFind uf(10); // 创建一个包含 10 个元素的并查集

    uf.unionSets(1, 2);
    uf.unionSets(2, 3);

    std::cout << "1 and 3 connected? " << (uf.find(1) == uf.find(3)) << std::endl; // 输出 1 和 3 是否连接

    return 0;
}
```
