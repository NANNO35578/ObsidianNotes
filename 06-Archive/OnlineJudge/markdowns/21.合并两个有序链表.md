[21. 合并两个有序链表](https://leetcode.cn/problems/merge-two-sorted-lists/)

##### 迭代
- 定义`dummy` 节点存储链表头
- 循环遍历list1与list2直到其中一个达到链表尾，将剩余部分加到合并后的结尾

```cpp
    ListNode* mergeTwoLists(ListNode* list1, ListNode* list2) {
        if (list1 == nullptr)
            return list2;
        if (list2 == nullptr)
            return list1;

        ListNode a = ListNode(-999);
        ListNode* dummy = &a;

        ListNode* point = nullptr;
        if (list1->val < list2->val) {
            dummy->next = list1;
            point=list1;
            list1=list1->next;
        } else {
            dummy->next = list2;
            point=list2;
            list2=list2->next;
        }

        while (list1 != nullptr && list2 != nullptr) {
            if (list1->val <= list2->val) {
                point->next = list1;
                list1 = list1->next;
                point = point->next;
            } else {
                point->next = list2;
                point = point->next;
                list2 = list2->next;
            }
        }

        if (list1 != nullptr) {
            point->next = list1;
        }
        if (list2 != nullptr) {
            point->next = list2;
        }
        return dummy->next;
    }
```

---
##### 递归


$$
\left\{
\begin{array}{lc}
list1[0]+merge(list1[1:],list2) & \text list1[0]<list2[0] \\
list1[0]+merge(list1[1:],list2) & \text otherwise
\end{array}
\right.
$$

也就是说，两个链表头部值较小的一个节点与剩下元素的 `merge` 操作结果合并。
```cpp
    ListNode* mergeTwoLists(ListNode* l1, ListNode* l2) {
        if (l1 == nullptr) {
            return l2;
        } else if (l2 == nullptr) {
            return l1;
        } else if (l1->val < l2->val) {
            l1->next = mergeTwoLists(l1->next, l2);
            return l1;
        } else {
            l2->next = mergeTwoLists(l1, l2->next);
            return l2;
        }
    }
```
