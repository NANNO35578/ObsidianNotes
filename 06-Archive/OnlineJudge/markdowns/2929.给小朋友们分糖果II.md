
[2929. 给小朋友们分糖果 II](https://leetcode.cn/problems/distribute-candies-among-children-ii/)


----

##### 思路一：动态规划 + 前缀和

- 令 $dp[i][j]$ 表示给前 $i$ 个孩子分配了 $j$ 颗糖的方案数。转移方程： $dp[i][j] =\sum_{k=0}^{\min(j,\;\text{limit})}$.
- 用前缀和可将每次转移的区间求和降为 $O(1)$：  

令$dp[i−1][t],pre[j]=\sum_{t=0}^j$ ,则 d$p[i][j]=pre[j] - (j-\text{limit}-1\ge0?\;pre[j-\text{limit}-1]:0)$

- 时间复杂度：$O(3\times n)$=O(n)  空间复杂度：$O(n)$

```cpp
long long distributeCandies(int n, int limit) {
    vector<ll> dp(n+1, 0), pre(n+1), nxt(n+1);
    dp[0] = 1;
    // 依次给 3 个孩子分
    for (int i = 1; i <= 3; i++) {
        // 计算 dp[i-1] 的前缀和
        pre[0] = dp[0];
        for (int j = 1; j <= n; j++)
            pre[j] = pre[j-1] + dp[j];
        // 用滑动窗口计算 dp[i][*]
        for (int j = 0; j <= n; j++) {
            ll total = pre[j];
            if (j - limit - 1 >= 0)
                total -= pre[j - limit - 1];
            nxt[j] = total;
        }
        swap(dp, nxt);
    }
    return dp[n];
}
```

##### 思路二：组合数学 + 容斥原理

- 记 $x_1,x_2,x_3\ge0$ 且 $x_1+x_2+x_3=n,\quad x_i\le\text{limit}.$

首先，不考虑上界时非负整数解数为$\binom{n+2}{2}$, 令事件 $A_i$ 表示第 $i$ 个孩子拿到 ≥ limit+1 颗糖，则用容斥：

$$\# = \binom{n+2}{2} -3\binom{n-(\text{limit}+1)+2}{2} +3\binom{n-2(\text{limit}+1)+2}{2} -\binom{n-3(\text{limit}+1)+2}{2}$$

其中，若组合数上下标不合法则视为 0

时间复杂度：$O(1)$，空间复杂度：$O(1)$

```cpp
// 计算 C(m, 2)，m<2 时返回 0
ll C2(ll m) {
    if (m < 2) return 0;
    return m * (m - 1) / 2;
}

long long distributeCandies(int n, int limit) {
    // C(n+2,2)
    ll total = C2((ll)n + 2);
    // -3 * C(n - limit - 1 + 2, 2) = -3 * C(n - limit + 1, 2)
    ll t1 = C2((ll)n - limit + 1);
    // +3 * C(n - 2*(limit+1) + 2, 2) = +3 * C(n - 2*limit, 2)
    ll t2 = C2((ll)n - 2LL * limit);
    // -1 * C(n - 3*(limit+1) + 2, 2) = -C(n - 3*limit - 1, 2)
    ll t3 = C2((ll)n - 3LL * limit - 1);
    return total - 3LL * t1 + 3LL * t2 - t3;
}
```
