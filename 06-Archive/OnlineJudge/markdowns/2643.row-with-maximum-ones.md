[一最多的行](https://leetcode.cn/problems/row-with-maximum-ones)
- 直接遍历
```cpp
class Solution {
public:
    vector<int> rowAndMaximumOnes(vector<vector<int>>& mat) {
       vector<int> ans{0, 0};
        int rowCnt = 0;
        int count = 0;
        for (vector<int>& row : mat) {
            count = 0;
            for (int i : row) count += i;
            if (count > ans[1]) {
                ans[0] = rowCnt;
                ans[1] = count;
            }
            ++rowCnt;
        }
        return ans;
    }
};
```

        int maxOnes = 0, rowIndex = 0;
        for (int i = 0; i < mat.size(); i++) {
            int tot = accumulate(mat[i].begin(), mat[i].end(), 0);
            if (tot > maxOnes) {
                maxOnes = tot;
                rowIndex = i;
            }
        }
        return {rowIndex, maxOnes};
