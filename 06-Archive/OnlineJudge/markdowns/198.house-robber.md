[打家劫舍](https://leetcode.cn/problems/house-robber/description/)
- 一维动态规划
- 状态转移方程 $dp[i]=max(dp[i−2]+nums[i],dp[i−1])$
- 可压缩至一维
```cpp
  int rob(vector<int>& nums) {
      if (nums.size() == 1)
          return nums[0];
      else if (nums.size() == 2)
          return nums[0] > nums[1] ? nums[0] : nums[1];
      int point = 2;// index
  
      int dNpre_2 = nums[0];
      int dNpre_1 = std::max(nums[0], nums[1]);
      int dN;
  
      while (point < nums.size()) {
          dN = std::max((dNpre_2 + nums[point]), dNpre_1);
          dNpre_2 = dNpre_1;
          dNpre_1 = dN;
          point++;
      }
      return dN;
  }
```
