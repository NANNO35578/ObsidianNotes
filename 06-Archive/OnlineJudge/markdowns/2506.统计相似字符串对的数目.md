[2506. 统计相似字符串对的数目](https://leetcode.cn/problems/count-pairs-of-similar-strings/)

- 标记字符串可使用一个int，若第i位不为0，则表示字符串包含该字符，使用`num|=(1<< ch- 'a');`操作即可
- 累计map中出现两次及以上次数的组合即可

```cpp
int similarPairs(std::vector<std::string>& words) {
    std::unordered_map<int, int> countMap;

    // 统计每个字符串的字符集合
    for (const std::string& word : words) {
        int flag = 0;
        for (char c : word) {
            flag |= (1 << (c - 'a')); // 使用位运算标记字符
        }
        countMap[flag]++;
    }

    int duplicateCount = 0;

    // 计算相似字符串对的数量
    for (const auto& entry : countMap) {
        int count = entry.second;
        if (count > 1) {
            duplicateCount += (count * (count - 1)) / 2; // 组合数 C(n, 2)
        }
    }

    return duplicateCount;
}
```


---
##### 优化
```cpp
    int similarPairs(vector<string>& words) {
        int res = 0;
        unordered_map<int, int> cnt;
        for (const string& word : words) {
            int state = 0;
            for (char c : word) {
                state |= 1 << (c - 'a');
            }
            res += cnt[state];
            cnt[state]++;
        }
        return res;
    }
```
