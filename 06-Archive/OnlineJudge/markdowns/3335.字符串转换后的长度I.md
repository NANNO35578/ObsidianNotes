[3335. 字符串转换后的长度 I](https://leetcode.cn/problems/total-characters-in-string-after-transformations-i/)
##### 递推

|**次数**|**转换后字符**|**函数**|**计算**|
|---|---|---|---|
|0|a|`fa_0`|0|
|1|b|`fa_1`|1|
|2|c|`fa_2`|2|
|3|d|`fa_3`|3|
|...|...|...|...|
|25|z|`fa_24`|26 * 0 + 25|
|26|ab|`fa_25 = fa_0 + fa_1`|26 * 1 + 0|
|27|bc|`fa_26 = fa_1 + fa_2`|26 * 1 + 1|
|28|cd|`fa_27 = fa_2 + fa_3`|26 * 1 + 2|
|...|...|...|...|
|50|yz|`fa_24 + fa_25`|26 * 1 + 24|
|51|zab|`fa_25 + fa_0 + fa_1`|26 * 1 + 25|
|52|abcd|`fa_52 = fa_25 + fa_28 = fa_0 + fa_1 + fa_2 + fa_3`|26 * 2 + 0|
|53|bcde|`fa_53 = fa_1 + fa_2 + fa_3 + fa_4`|26 * 2 + 1|
```cpp
    int lengthAfterTransformations(string s, int t) {
        const int MOD = 1e9 + 7;
        vector<int> f(26, 0);
        
        // 初始化 f(0, c)
        for (char ch : s) {
            f[ch - 'a']++;
        }
        
        // 进行 t 次转换
        for (int i = 0; i < t; ++i) {
            vector<int> newF(26, 0);
            newF[0] = f[25]; // a 可以由 z 转换而来
            newF[1] = (f[25] + f[0]) % MOD;
            for (int c = 2; c < 26; ++c) {
                newF[c] = f[c - 1]; // 其他字符
            }
            f = newF; // 更新 f
        }
        
        // 计算最终长度
        long long totalLength = 0;
        for (int count : f) {
            totalLength = (totalLength + count) % MOD;
        }
        
        return totalLength;
    }
```

##### dp
- **初始化**:
    
    - `vector<int> dp(26, 1)`：初始假设每个字符在0次转换时都出现一次。
- **动态规划填充**:
    
    - 使用循环从 26 到 `t + 25` 填充 `dp` 数组：
        - `dp[i % 26] = (dp[(i - 26) % 26] + dp[(i - 25) % 26]) % mod;`
        - 这条公式利用了前两个状态来计算当前状态，模拟字符转换的过程。
- **计算结果**:
    
    - 遍历字符串 `s`，对每个字符 `i`:
        - 利用 `dp[(t + i - 'a') % 26]` 获取经过 `t` 次转换后字符的出现次数。
        - 将结果累加到 `ans` 中，并对 `mod` 取余。
- **返回结果**:
    
    - 最终返回 `ans`，即字符串在经过 `t` 次转换后的总长度。

```cpp
class Solution {
public:
    int lengthAfterTransformations(string s, int t) {
        int mod = 1e9 + 7;
        vector<int> dp(26, 1);
        for (int i = 26; i <= t + 25; i++) {
            dp[i % 26] = (dp[(i - 26) % 26] + dp[(i - 25) % 26]) % mod;
        }
        long long ans = 0;
        for (char i : s) {
            ans = (ans + dp[(t + i - 'a') % 26]) % mod;
        }
        return ans;
    }
};
```
