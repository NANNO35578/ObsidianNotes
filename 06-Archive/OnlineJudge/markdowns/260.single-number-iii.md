[只出现一次的数字3](https://leetcode.cn/problems/single-number-iii/description/)
_此题防止溢出可使用`unsigned int`_
- 由位异或可轻松得到 a\^b 而利用通过`aXb & (-aXb)`找到其中一个数字的最右边的1位，称为`separate_mark`。这位的存在可以帮助我们将数组分为两组。然后再与a\^b 进行亦或即可得到其中一个数
```cpp
class Solution {
public:
    vector<int> singleNumber(vector<int>& nums) {
        vector<int> ans{0, 0};
        // 第一步：计算数组中所有元素的异或值，得到num1_Xor_num2，即这两个只出现一次的数字（num1和num2）的异或值。
        unsigned int aXb = nums[0];// 官方防止溢出: int lsb = (xorsum == INT_MIN ? xorsum : xorsum & (-xorsum));
        for (int i = 1; i < nums.size(); ++i) aXb ^= nums[i];
        // 第二步：通过num1_Xor_num2找到其中一个数字的最右边的1位，称为separate_mark。这位的存在可以帮助我们将数组分为两组。
        int spBit = aXb & (-aXb);
        // 第三步：遍历数组，将所有包含separate_mark位的数字进行异或运算，得到其中一个数字（resA）。
        for (int i : nums)
            if (spBit & i) ans[0] ^= i;
        // 第四步：通过resA和num1_Xor_num2的异或运算，可以得到另一个数字（resB）。
        ans[1] = ans[0] ^ aXb;
        return ans;
    }
};
```
# 方法二：哈希表
- 可以使用一个哈希映射统计数组中每一个元素出现的次数。
- 在统计完成后，我们对哈希映射进行遍历，将所有只出现了一次的数放入答案中。
```cpp
class Solution {
public:
    vector<int> singleNumber(vector<int>& nums) {
        unordered_map<int, int> freq;
        for (int num: nums) {
            ++freq[num];
        }
        vector<int> ans;
        for (const auto& [num, occ]: freq) {
            if (occ == 1) {
                ans.push_back(num);
            }
        }
        return ans;
    }
};
```
