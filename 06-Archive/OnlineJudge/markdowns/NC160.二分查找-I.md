[二分查找-I](http://nowcoder.com/profile/386104436/codeBookDetail?submissionId=456270965)
- 注意处理值为左右点
```cpp
 int search(vector<int>& nums, int target) {
     // write code here
     if (nums.empty()) return -1;
     int lp = 0;
     int rp = nums.size() - 1;
     int p = (lp + rp) >> 1;
     while (nums[p] != target) {
         if (nums[p] < target) {
             if (nums[rp] == target) return rp;
             lp = p;
         } else {
             if (nums[lp] == target) return lp;
             rp = p;
         }
         p = (lp + rp) >> 1;
         if (lp + 1 == rp || lp == rp) return -1;
     }
     return p;
 }
```
