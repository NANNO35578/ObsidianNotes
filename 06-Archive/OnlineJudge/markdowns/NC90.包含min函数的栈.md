[包含min函数的栈](https://www.nowcoder.com/profile/386104436/codeBookDetail?submissionId=457590685)
- 使用一个栈来存储最小值
- 新来的元素若小于栈顶则存入栈顶 否则重复存入栈顶
```cpp
class Solution {
   private:
    std::stack<int> minStack;
    std::stack<int> orgStack;

   public:
    void push(int value) {
        if (orgStack.empty()) {
            minStack.push(value);
            orgStack.push(value);
            return;
        }
        orgStack.push(value);
        if (value < this->min()) {
            minStack.push(value);
        } else {
            minStack.push(this->min());
        }
    }
    void pop() {
        minStack.pop();
        orgStack.pop();
    }
    int top() { return orgStack.top(); }
    int min() { return minStack.top(); }
};
```
