[1550. 存在连续三个奇数的数组](https://leetcode.cn/problems/three-consecutive-odds/)

```cpp
    bool threeConsecutiveOdds(vector<int>& arr) {
	    int count = 0;
	    for (int num : arr) {
	        if (num % 2 != 0) { // 判断是否为奇数
	            count++;
	            if (count == 3) {
	                return true; // 找到三个连续的奇数
	            }
	        } else {
	            count = 0; // 重置计数
	        }
	    }
	    return false; // 没有找到连续三个奇数
    }
```
