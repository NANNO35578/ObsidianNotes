    bool isValid(string s) {
        // write code here
        stack<char> pa;
        for (char ch : s) {
            if (pa.empty()) {
                pa.push(ch);
                continue;
            }
            if (ch == ')' && pa.top() == '(') {
                pa.pop();
            } else if (ch == ']' && pa.top() == '[') {
                pa.pop();
            } else if (ch == '}' && pa.top() == '{') {
                pa.pop();
            } else {
                pa.push(ch);
            }
        }
        return pa.empty();
    }
