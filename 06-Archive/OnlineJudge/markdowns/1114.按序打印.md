[1114. 按序打印](https://leetcode.cn/problems/print-in-order/)

##### C语言条件变量，互斥锁
- 使用循环等待（while 循环）是为了处理虚假唤醒（spurious wakeup），确保线程在被唤醒后仍然检查条件
```cpp
#include <pthread.h>
#include <functional>

class Foo {
private:
    pthread_cond_t condFirst;
    pthread_cond_t condSecond;
    pthread_mutex_t lock;
    int order;

public:
    Foo() : order(0) {
        pthread_cond_init(&condFirst, NULL);
        pthread_cond_init(&condSecond, NULL);
        pthread_mutex_init(&lock, NULL);
    }

    void first(std::function<void()> printFirst) {
        pthread_mutex_lock(&lock);
        printFirst(); // 打印 "first"
        order = 1;
        pthread_cond_broadcast(&condSecond); // 唤醒等待 second 的线程
        pthread_mutex_unlock(&lock);
    }

    void second(std::function<void()> printSecond) {
        pthread_mutex_lock(&lock);
        while (order < 1) { // 循环等待
            pthread_cond_wait(&condSecond, &lock);
        }
        printSecond(); // 打印 "second"
        order = 2;
        pthread_cond_broadcast(&condFirst); // 唤醒等待 third 的线程
        pthread_mutex_unlock(&lock);
    }

    void third(std::function<void()> printThird) {
        pthread_mutex_lock(&lock);
        while (order < 2) { // 循环等待
            pthread_cond_wait(&condFirst, &lock);
        }
        printThird(); // 打印 "third"
        pthread_mutex_unlock(&lock);
    }
};
```

##### 信号量（semaphore）

```cpp
#include <semaphore.h>

class Foo {

protected:
    sem_t firstJobDone;
    sem_t secondJobDone;

public:
    Foo() {
        sem_init(&firstJobDone, 0, 0);
        sem_init(&secondJobDone, 0, 0);
    }

    void first(function<void()> printFirst) {
        // printFirst() outputs "first".
        printFirst();
        sem_post(&firstJobDone);
    }

    void second(function<void()> printSecond) {
        sem_wait(&firstJobDone);
        // printSecond() outputs "second".
        printSecond();
        sem_post(&secondJobDone);
        
    }

    void third(function<void()> printThird) {
        sem_wait(&secondJobDone);
        // printThird() outputs "third".
        printThird();
    }
};
```

##### C++条件变量（`condition_variable`）和互斥锁（`mutex`）
- **顺序控制**：利用一个共享变量 `k` 来控制函数的执行顺序。`first`、`second` 和 `third` 的执行依赖于 `k` 的值
- **阻塞与唤醒**：使用条件变量和互斥锁实现线程的阻塞和唤醒机制，确保在适当的时机唤醒等待的线程
- **线程安全**：通过 `unique_lock` 和 `mutex`，确保对共享资源的安全访问，避免数据竞

```cpp
#include<thread>
#include<condition_variable>
#include<mutex>
using namespace std;

class Foo {
    condition_variable cv;
    mutex mtx;
    int k = 0;

public:
    Foo() { }

    void first(function<void()> printFirst) {
        // printFirst() outputs "first". Do not change or remove this line.
        printFirst();
        k = 1;
        cv.notify_all();
    }

    void second(function<void()> printSecond) {
        // printSecond() outputs "second". Do not change or remove this line.
        unique_lock<mutex> lock(mtx);
        cv.wait(lock, [this](){return k == 1; });
        printSecond();
        k = 2;
        cv.notify_one();
    }

    void third(function<void()> printThird) {
        unique_lock<mutex> lock(mtx);
        cv.wait(lock,[this](){return k == 2;});
        // printThird() outputs "third". Do not change or remove this line.
        printThird();
    }
};
```
