[2359. 找到离给定两个节点最近的节点](https://leetcode.cn/problems/find-closest-node-to-given-two-nodes/)
- **定义一个函数** `bfs`，用于从指定的起始节点进行广度优先搜索，返回每个节点的距离。
- **计算 `node1` 和 `node2` 到所有其他节点的距离**。
- **遍历所有节点**，找到距离的较大值最小的节点。
- **处理无解情况**，返回 `-1`。

```cpp
vector<int> bfs(const vector<int>& edges, int start) {
    int n = edges.size();
    vector<int> distance(n, numeric_limits<int>::max());
    queue<int> q;
    distance[start] = 0;
    q.push(start);
    
    while (!q.empty()) {
        int node = q.front();
        q.pop();
        
        int next = edges[node];
        if (next != -1 && distance[next] == numeric_limits<int>::max()) {
            distance[next] = distance[node] + 1;
            q.push(next);
        }
    }
    
    return distance;
}

int closestMeetingNode(vector<int>& edges, int node1, int node2) {
    vector<int> dist1 = bfs(edges, node1);
    vector<int> dist2 = bfs(edges, node2);
    
    int bestNode = -1;
    int minMaxDistance = numeric_limits<int>::max();
    
    for (int i = 0; i < edges.size(); ++i) {
        int d1 = dist1[i];
        int d2 = dist2[i];
        
        if (d1 != numeric_limits<int>::max() && d2 != numeric_limits<int>::max()) {
            int maxDistance = max(d1, d2);
            if (maxDistance < minMaxDistance || (maxDistance == minMaxDistance && i < bestNode)) {
                minMaxDistance = maxDistance;
                bestNode = i;
            }
        }
    }
    
    return bestNode;
}
```

##### 优化
```cpp
class Solution {
public:
    int closestMeetingNode(vector<int>& edges, int node1, int node2) {
        int n = edges.size();
        auto get = [&](int x) {
            vector<int> d(n, -1);
            int p = x;
            int dis = 0;
            while (p != -1 && d[p] == -1) {
                d[p] = dis;
                p = edges[p];
                dis++;
            }
            return d;
        };
        auto d1 = get(node1);
        auto d2 = get(node2);
        int res = -1;
        for (int i = 0; i < n; i++) {
            if (d1[i] != -1 && d2[i] != -1 && (res == -1 || max(d1[res], d2[res]) > max(d1[i], d2[i]))) {
                res = i;
            }
        }
        return res;
    }
};
```
