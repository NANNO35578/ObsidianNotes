[组合](https://leetcode.cn/problems/combinations/description/)
- 排列回溯的是交换的位置，而组合回溯的*是否*把当前的数字加入结果中
```cpp
    // 主函数
    vector<vector<int>> combine(int n, int k) {
        vector<vector<int>> ans;
        vector<int> comb(k, 0);
        int count = 0;
        backtracking(ans, comb, count, 1, n, k);
        return ans;
    }
    // 辅函数
    void backtracking(vector<vector<int>>& ans, vector<int>& comb, int& count,
                      int pos, int n, int k) {
        if (count == k) {
            ans.push_back(comb);
            return;
        }
        for (int i = pos; i <= n; ++i) {
            comb[count++] = i;                            // 修改当前节点状态
            backtracking(ans, comb, count, i + 1, n, k);  // 递归子节点
            --count;
        }
    }
```

> 优化后的代码

```cpp
    vector<int> temp;
    vector<vector<int>> ans;

    void dfs(int cur, int n, int k) {
        // 剪枝：temp 长度加上区间 [cur, n] 的长度小于 k，不可能构造出长度为 k 的 temp
        if (temp.size() + (n - cur + 1) < k) {
            return;
        }
        // 记录合法的答案
        if (temp.size() == k) {
            ans.push_back(temp);
            return;
        }
        // 考虑选择当前位置
        temp.push_back(cur);
        dfs(cur + 1, n, k);
        temp.pop_back();
        // 考虑不选择当前位置
        dfs(cur + 1, n, k);
    }

    vector<vector<int>> combine(int n, int k) {
        dfs(1, n, k);
        return ans;
    }

```
