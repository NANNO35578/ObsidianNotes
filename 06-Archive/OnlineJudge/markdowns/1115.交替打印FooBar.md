[1115. 交替打印 FooBar](https://leetcode.cn/problems/print-foobar-alternately/)

##### 信号量
- bar线程先发送一次barSem
- foo线程进入循环等待barSem
- 打印后发送fooSem
- bar线程进入循环，等待fooSem
- 打印完发送barSem

```cpp
class FooBar {
private:
    int n;
    sem_t barSem;
    sem_t fooSem;

public:
    FooBar(int n) {
        this->n = n;

        sem_init(&fooSem, 0, 0);
        sem_init(&barSem, 0, 0);
    }

    void foo(function<void()> printFoo) {
        for (int i = 0; i < n; i++) {
            sem_wait(&barSem);
            // printFoo() outputs "foo". Do not change or remove this line.
            printFoo();
            sem_post(&fooSem);
        }
    }

    void bar(function<void()> printBar) {
        sem_post(&barSem);
        for (int i = 0; i < n; i++) {
            sem_wait(&fooSem);
            // printBar() outputs "bar". Do not change or remove this line.
            printBar();
            sem_post(&barSem);
        }
    }
};

```

##### 互斥锁
- 使用两个mutex lock来序列化交替打印Foo， Bar， Foo Bar　．．．

```cpp
class FooBar {
private:
    int n;
    // Use mtx1 and mtx2 to serialize printFoo and printBar alternatively
    mutex mtx1, mtx2;

public:
    FooBar(int n) {
        this->n = n;
        // so printBar() can never print since it need acquire the mtx2 lock first
        mtx2.lock();
    }

    void foo(function<void()> printFoo) {
        
        for (int i = 0; i < n; i++) {
            // only preceed to printFoo if acquire mtx1 lock
            mtx1.lock();
        	printFoo();
            mtx2.unlock();
        }
    }

    void bar(function<void()> printBar) {
        
        for (int i = 0; i < n; i++) {
            // only preceed to printBar if acquire mtx1 lock
            mtx2.lock();
        	printBar();
            mtx1.unlock();
        }
    }
};
```
