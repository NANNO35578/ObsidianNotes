[49. 字母异位词分组](https://leetcode.cn/problems/group-anagrams/)

- 根据string中字符个数创建26长度vector
- 根据vector创建键，值存储字符串
- 根据相同的键，将值存入同一纬度

```cpp
    vector<int> getNum(string s) {
        vector<int> ret(26, 0);
        for (char c : s) {
            ret[c - 'a']++;
        }
        return ret;
    }

    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        unordered_map<string, vector<string>> anagramMap;

        for (string& s : strs) {
            vector<int> num = getNum(s);
            // 将向量转换为字符串，以便用作 map 的键
            string key = "";
            for (int count : num) {
                key += to_string(count) + ",";
            }
            anagramMap[key].push_back(s);
        }

        vector<vector<string>> result;
        for (const auto& pair : anagramMap) {
            result.push_back(pair.second);
        }

        return result;
    }
```

##### 优化
- 直接将string转换为键

```cpp
    string getNumStr(string s) {
        string ret(26, 0);
        for (char c : s) {
            ret[c - 'a']++;
        }
        return ret;
    }

    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        unordered_map<string, vector<string>> anagramMap;

        for (string& s : strs) {
            anagramMap[getNumStr(s)].push_back(s);
        }

        vector<vector<string>> result;
        for (const auto& pair : anagramMap) {
            result.push_back(pair.second);
        }

        return result;
    }
```


---

- 由于互为字母异位词的两个字符串包含的字母相同，因此两个字符串中的相同字母出现的次数一定是相同的，故可以将每个字母出现的次数使用字符串表示，作为哈希表的键。

**哈希函数**：
- 自定义一个哈希函数 `arrayHash`，用于对字符频率统计的数组（`array<int, 26>`）进行哈希。这个数组的每个元素表示对应字母出现的次数。

```cpp
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        // 自定义对 array<int, 26> 类型的哈希函数
        auto arrayHash = [fn = hash<int>{}] (const array<int, 26>& arr) -> size_t {
            return accumulate(arr.begin(), arr.end(), 0u, [&](size_t acc, int num) {
                return (acc << 1) ^ fn(num);
            });
        };

        unordered_map<array<int, 26>, vector<string>, decltype(arrayHash)> mp(0, arrayHash);
        for (string& str: strs) {
            array<int, 26> counts{};
            int length = str.length();
            for (int i = 0; i < length; ++i) {
                counts[str[i] - 'a'] ++;
            }
            mp[counts].emplace_back(str);
        }
        vector<vector<string>> ans;
        for (auto it = mp.begin(); it != mp.end(); ++it) {
            ans.emplace_back(it->second);
        }
        return ans;
    }
```
