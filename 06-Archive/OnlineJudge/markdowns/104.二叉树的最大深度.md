[104. 二叉树的最大深度](https://leetcode.cn/problems/maximum-depth-of-binary-tree/)

- 层序遍历 广搜
- 用空节点分隔每一层


```cpp
    int maxDepth(TreeNode* root) {
        // 特殊情况
        if (root == nullptr) return 0;
        if (root->left == nullptr && root->right == nullptr)
            return 1;

		int cnt=0;
        // 队列
        queue<TreeNode *> levelTree = {};
        // 首节点
        levelTree.push(root);
        levelTree.push(nullptr);
        ++cnt;
        // 队列只剩下标记
        while (levelTree.size() > 1) {
            TreeNode *head = levelTree.front();
            // 取出队首
            /// 队首不为标记
            if (head != nullptr) {
                // 把孩子插入队尾
                if (head->left != nullptr) levelTree.push(head->left);
                if (head->right != nullptr) levelTree.push(head->right);
                // 存储队首val
            } else {
                /// 队首为标记
                // 将该层存入答案 该层vec清0
                // 添加标记
                levelTree.push(nullptr);
                ++cnt;
            }
            // 弹出队首
            levelTree.pop();
        }
        return cnt;
    }
```

##### dfs

```cpp
    int maxDepth(TreeNode* root) {
        if (root == nullptr) return 0;
        return max(maxDepth(root->left), maxDepth(root->right)) + 1;
    }
```
