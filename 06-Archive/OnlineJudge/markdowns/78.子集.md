[78. 子集](https://leetcode.cn/problems/subsets/)

##### 回溯
- **递归回溯**：从数组的每个元素开始，可以选择包含或不包含该元素，直到遍历完所有元素
- **存储结果**：每次选择一个元素后，递归调用函数，直到到达数组的末尾。每次到达末尾时，将当前的子集存入结果中
具体流程
- **`backtrack` 函数**：
    - 将当前子集添加到结果集
    - 循环遍历从 `start` 到 `nums.size()` 的每个元素：
        - 将当前元素添加到 `currentSubset`
        - 递归调用 `backtrack`，更新 `start` 为 `i + 1`（避免重复选择同一元素）
        - 使用 `pop_back` 撤销选择，准备回溯到上一个状态

```cpp
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> result;
        vector<int> currentSubset;
        backtrack(nums, 0, currentSubset, result);
        return result;
    }

    void backtrack(vector<int>& nums, int start, vector<int>& currentSubset, vector<vector<int>>& result) {
        // 每次调用后，当前子集都要加入结果
        result.push_back(currentSubset);

        for (int i = start; i < nums.size(); ++i) {
            // 选择当前元素
            currentSubset.push_back(nums[i]);
            // 递归调用，开始选择下一个元素
            backtrack(nums, i + 1, currentSubset, result);
            // 撤销选择，准备下一个子集
            currentSubset.pop_back();
        }
    }
```

##### 迭代
- 对于数组中的每个元素，遍历当前的结果集，将该元素添加到每个已有的子集中并存入结果集

```cpp
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> result = {{}};
        
        for (int num : nums) {
            // 对已有的子集进行扩展
            int currentSize = result.size();
            for (int i = 0; i < currentSize; ++i) {
                vector<int> newSubset = result[i];
                newSubset.push_back(num);
                result.push_back(newSubset);
            }
        }
        
        return result;
    }
```

##### 位运算
- 每个子集可以用一个二进制数表示，其中每一位的值表示是否包含对应位置的元素
```cpp
    vector<vector<int>> subsets(vector<int>& nums) {
        int n = nums.size();
        vector<vector<int>> result;
        
        // 2^n 个子集
        for (int i = 0; i < (1 << n); ++i) {
            vector<int> subset;
            for (int j = 0; j < n; ++j) {
                // 检查第 j 位是否为 1
                if (i & (1 << j)) {
                    subset.push_back(nums[j]);
                }
            }
            result.push_back(subset);
        }
        
        return result;
    }
```
