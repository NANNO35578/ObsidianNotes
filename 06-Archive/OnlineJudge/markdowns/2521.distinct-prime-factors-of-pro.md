[数组乘积的不同质因数](https://leetcode.cn/problems/distinct-prime-factors-of-product-of-array/description/) 
- **计算乘积**：虽然直接计算乘积可能会导致数值溢出，但我们可以通过分解每个数来避免这个问题。
- **质因数分解**：对每个数进行质因数分解，记录所有出现的质因数。
- **去重**：使用集合来存储质因数，以确保每个质因数只计算一次。
```cpp
   std::set<int> primeFactors(int n) {
        std::set<int> factors;

        // 处理2这个质因数
        while (n % 2 == 0) {
            factors.insert(2);
            n /= 2;
        }

        // 处理其他质因数
        for (int i = 3; i <= std::sqrt(n); i += 2) {
            while (n % i == 0) {
                factors.insert(i);
                n /= i;
            }
        }

        if (n > 2) {
            factors.insert(n);
        }

        return factors;
    }
    int distinctPrimeFactors(vector<int>& nums) {
        std::set<int> distinctPrimes;

        for (int num : nums) {
            std::set<int> factors = primeFactors(num);
            distinctPrimes.insert(factors.begin(), factors.end());
        }

        return distinctPrimes.size();
    }
```
##### 使用map统计所有数字的质因数，最后的答案就是`cnt.size()`
```cpp
    int distinctPrimeFactors(vector<int>& nums) {
        map<int,int>cnt;
        for(int i = 0; i < nums.size(); i++){
            for(int j = 2; j <= nums[i]; j++){
                while(nums[i] % j == 0){
                    cnt[j]++;
                    nums[i] /= j;
                }
            }
        }
        return cnt.size();
    }
```
