[56. 合并区间](https://leetcode.cn/problems/merge-intervals/)


##### 排序 模拟
- 对区间按照起始从小到大，相同末尾从小到大排序

```cpp
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        if (intervals.size() == 1)
            return intervals;
        sort(intervals.begin(), intervals.end(),
             [](vector<int>& a, vector<int>& b) {
                 if (a[0] == b[0])
                     return a[1] < b[1];
                 return a[0] < b[0];
             });
        vector<vector<int>> ans;
        int l = intervals[0][0];
        int r = intervals[0][1];

        for (int i = 1; i < intervals.size(); ++i) {

            if (r < intervals[i][0]) {
                ans.push_back(vector<int>{l, r});
                l = intervals[i][0];
                r = intervals[i][1];
            } else if (r >= intervals[i][0] && intervals[i][1] > r) {
                r = intervals[i][1];
            }
        }
        ans.push_back(vector<int>{l, r});
        return ans;
    }
```

- *优化版本*
```cpp
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        sort(intervals.begin(), intervals.end(),
             [](const vector<int>& a, const vector<int>& b) -> bool {
                 return a[0] < b[0];
             });
        vector<vector<int>> rt{intervals[0]};
        for (auto it = intervals.begin() + 1; it != intervals.end(); ++it) {
            auto& merged = rt.back();
            if (merged[1] >= (*it)[0]) {
                merged[1] = max(merged[1], (*it)[1]);
            } else {
                rt.emplace_back(*it);
            }
        }
        return rt;
    }
```
