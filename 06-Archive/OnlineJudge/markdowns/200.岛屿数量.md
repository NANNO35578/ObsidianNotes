1. **初始化**：
    - 创建一个计数器 `islandCount` 用于记录岛屿的数量。
    - 遍历整个网格，检查每个单元格。
2. **遍历网格**：
    - 如果遇到 '1'（陆地），则说明找到了一个新的岛屿，计数器 `islandCount` 加一。
    - 然后从这个单元格开始进行 DFS 或 BFS，标记所有与之相连的 '1' 为 '0'（水），以避免重复计数。
3. **DFS/BFS 实现**：
    - 定义一个递归函数（或使用队列实现 BFS），该函数会检查当前单元格的上下左右四个方向。
    - 如果相邻的单元格是 '1'，则继续调用该函数，直到所有相连的 '1' 都被标记为 '0'。

```cpp
    int numIslands(vector<vector<char>>& grid) {
        if (grid.empty()) return 0;
        
        int rows = grid.size();
        int cols = grid[0].size();
        int islandCount = 0;

        for (int i = 0; i < rows; ++i) {
            for (int j = 0; j < cols; ++j) {
                if (grid[i][j] == '1') {
                    // 找到一个新的岛屿
                    islandCount++;
                    // 使用 DFS 标记所有相连的陆地
                    dfs(grid, i, j);
                }
            }
        }

        return islandCount;
    }
 
    void dfs(vector<vector<char>>& grid, int i, int j) {
        // 边界条件
        if (i < 0 || i >= grid.size() || j < 0 || j >= grid[0].size() || grid[i][j] == '0') {
            return;
        }
        
        // 标记为水
        grid[i][j] = '0';

        // 递归上下左右
        dfs(grid, i + 1, j);
        dfs(grid, i - 1, j);
        dfs(grid, i, j + 1);
        dfs(grid, i, j - 1);
    }
```

##### 迭代
1. **使用迭代 DFS**：使用栈而不是递归来实现深度优先搜索，避免可能的栈溢出问题，尤其是在处理较大网格时
2. **合并方向数组**：使用方向数组来简化代码，减少重复的代码行

```cpp
    int numIslands(vector<vector<char>>& grid) {
        if (grid.empty()) return 0;

        int rows = grid.size();
        int cols = grid[0].size();
        int islandCount = 0;

        // 定义方向数组
        vector<pair<int, int>> directions = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};

        for (int i = 0; i < rows; ++i) {
            for (int j = 0; j < cols; ++j) {
                if (grid[i][j] == '1') {
                    // 找到一个新的岛屿
                    islandCount++;
                    // 使用迭代 DFS
                    stack<pair<int, int>> s;
                    s.push({i, j});

                    while (!s.empty()) {
                        auto [x, y] = s.top();
                        s.pop();
                        grid[x][y] = '0'; // 标记为水

                        // 遍历所有方向
                        for (const auto& dir : directions) {
                            int newX = x + dir.first;
                            int newY = y + dir.second;
                            if (newX >= 0 && newX < rows && newY >= 0 && newY < cols && grid[newX][newY] == '1') {
                                s.push({newX, newY});
                            }
                        }
                    }
                }
            }
        }

        return islandCount;
    }
```
