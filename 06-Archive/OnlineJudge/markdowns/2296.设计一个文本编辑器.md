[2296. 设计一个文本编辑器](https://leetcode.cn/problems/design-a-text-editor/) 


##### *数据结构选择*
- **字符串 (`string currentText`)**：用于存储当前编辑的文本内容
- **整型 (`int cursorPosition`)**：用于跟踪光标在文本中的位置

##### *功能实现*
`addText` 
   - 使用 `insert` 方法在 `cursorPosition` 处插入 `text`
   - 更新 `cursorPosition` 为新文本的结束位置，即 `cursorPosition += text.length()`
`deleteText`
   - 计算实际能删除的字符数 `actualDeleteCount`，为 `min(k, cursorPosition)`
   - 使用 `erase` 方法删除光标左侧的字符
   - 更新 `cursorPosition`，确保其不小于 0
`cursorLeft` 和 `cursorRight`
   - 在 `cursorLeft` 中，更新光标位置为 `max(0, cursorPosition - k)`
   - 在 `cursorRight` 中，更新光标位置为 `min(currentText.length(), cursorPosition + k)`
   - 使用 `substr` 方法返回光标左侧最大 10 个字符



```cpp
class TextEditor {
private:
    string currentText; // 当前文本
    int cursorPosition; // 光标位置

public:
    // 构造函数，初始化为空文本
    TextEditor() {
        currentText = "";
        cursorPosition = 0;
    }
    
    // 在光标所在位置添加文本
    void addText(string text) {
        currentText.insert(cursorPosition, text); // 在光标位置插入文本
        cursorPosition += text.length(); // 更新光标位置
    }
    
    // 删除光标左边 k 个字符，返回实际删除的字符数
    int deleteText(int k) {
        int actualDeleteCount = min(k, cursorPosition); // 计算实际删除的字符数
        currentText.erase(cursorPosition - actualDeleteCount, actualDeleteCount); // 删除字符
        cursorPosition -= actualDeleteCount; // 更新光标位置
        return actualDeleteCount;
    }
    
    // 将光标向左移动 k 次，返回光标左边的字符
    string cursorLeft(int k) {
        cursorPosition = max(0, cursorPosition - k); // 移动光标
        return currentText.substr(max(0, cursorPosition - 10), min(10, cursorPosition)); // 返回光标左边的字符
    }
    
    // 将光标向右移动 k 次，返回光标左边的字符
    string cursorRight(int k) {
        cursorPosition = min((int)currentText.length(), cursorPosition + k); // 移动光标
        return currentText.substr(max(0, cursorPosition - 10), min(10, cursorPosition)); // 返回光标左边的字符
    }
};
```

##### 双向链表
- 使用双向链表保存文本编辑器的文本，同时使用 cursor 指向处于光标右侧的链表节点。为了方便处理，初始时往链表中插入一个哨兵节点，该节点不保存文本字符，那么 cursor 指向该节点。
四种方法的处理如下：
	`addText`：遍历 text 字符串，将当前遍历的字符插入到 cursor 的前面
	`deleteText`：使用 count 记录删除的节点数目，不断地删除 cursor 前面的节点直到删除的节点数目等于 k 或 cursor 前面的节点为空
	`cursorLeft`：不断地将 cursor 移动到前面的节点直到移动次数等于 k 或 cursor 前面的节点为空。使用 head 指向 cursor 前面第 10 个节点或首节点，将 head 与 cursor 之间的字符返回（不包含 cursor）
	`cursorRight`：不断地将 cursor 移动到后面的节点直到移动次数等于 k 或 cursor 后面的节点为空。使用 head 指向 cursor 前面第 10 个节点或首节点，将 head 与 cursor 之间的字符返回（不包含 cursor）

```cpp
class TextEditor {
private:
    list<char> editor;
    list<char>::iterator cursor;

public:
    TextEditor() {
        cursor = editor.end();
    }

    void addText(string text) {
        for (char c : text) {
            editor.insert(cursor, c);
        }
    }

    int deleteText(int k) {
        int count = 0;
        for (; k > 0 && cursor != editor.begin(); k--) {
            editor.erase(prev(cursor));
            count++;
        }
        return count;
    }

    string cursorLeft(int k) {
        while (k > 0 && cursor != editor.begin()) {
            k--;
            cursor = prev(cursor);
        }
        auto head = cursor;
        for (int i = 0; i < 10 && head != editor.begin(); i++) {
            head = prev(head);
        }
        return string(head, cursor);
    }

    string cursorRight(int k) {
        while (k > 0 && cursor != editor.end()) {
            k--;
            cursor = next(cursor);
        }
        auto head = cursor;
        for (int i = 0; i < 10 && head != editor.begin(); i++) {
            head = prev(head);
        }
        return string(head, cursor);
    }
};
```

##### 栈思想
- 实际上使用`std::vector`,因为可以读取栈底部元素
- 同时多次删除添加减少空间申请与释放

```cpp
class TextEditor {
private:
    vector<char> left;
    vector<char> right;
public:
    TextEditor() {
        
    }

    void addText(string text) {
        for (char c : text) {
            left.push_back(c);
        }
    }
    
    int deleteText(int k) {
        int n = 0;
        for (; !left.empty() && k > 0; k--) {
            left.pop_back();
            n++;
        }
        return n;
    }
    
    string cursorLeft(int k) {
        while (!left.empty() && k > 0) {
            right.push_back(left.back());
            left.pop_back();
            k--;
        }
        int n = left.size();
        return string(left.begin() + max(0, n - 10), left.end());
    }
    
    string cursorRight(int k) {
        while (!right.empty() && k > 0) {
            left.push_back(right.back());
            right.pop_back();
            k--;
        }
        int n = left.size();
        return string(left.begin() + max(0, n - 10), left.end());
    }
};
```
