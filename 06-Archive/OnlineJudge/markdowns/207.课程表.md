[207. 课程表](https://leetcode.cn/problems/course-schedule/) 

- 每门课程可以看作一个节点，先修课程的关系可以看作有向边
- 测这个有向图是否存在环，如果存在环，则无法完成所有课程

##### DFS
1. **构建图**：使用邻接表表示图。
2. **使用状态数组**：定义一个状态数组，记录每个节点的状态：
    - 0: 未访问
    - 1: 正在访问（在递归栈中）
    - 2: 已访问（已完成探索）
3. **DFS 遍历**：对每个未访问的节点进行 DFS，检查是否存在环。

```cpp
bool dfs(int course, vector<vector<int>>& graph, vector<int>& visited) {
    if (visited[course] == 1) return false; // 当前节点正在访问，发现环
    if (visited[course] == 2) return true;  // 已访问，跳过

    // 标记当前节点为正在访问
    visited[course] = 1;

    // 访问所有相邻节点
    for (int next : graph[course]) {
        if (!dfs(next, graph, visited)) {
            return false; // 有环
        }
    }

    // 标记当前节点为已访问
    visited[course] = 2;
    return true;
}

bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
    vector<vector<int>> graph(numCourses);
    for (const auto& pre : prerequisites) {
        graph[pre[1]].push_back(pre[0]); // 先修边
    }

    vector<int> visited(numCourses, 0); // 0: 未访问, 1: 正在访问, 2: 已访问

    for (int i = 0; i < numCourses; ++i) {
        if (visited[i] == 0) {
            if (!dfs(i, graph, visited)) {
                return false; // 存在环
            }
        }
    }

    return true; // 可完成所有课程
}
```

##### 拓扑排序
- 用于有向无环图（DAG）的排序方法，可以用来解决课程安排的问题

### 拓扑排序的思路
1. **构建图**：使用邻接表表示课程之间的先修关系。
2. **计算入度**：为每个课程计算入度，入度表示有多少课程需要在当前课程之前完成。
3. **使用队列**：将所有入度为 0 的课程加入队列，表示这些课程可以直接学习。
4. **处理课程**：
    - 从队列中取出课程，减少其后继课程的入度。
    - 如果后继课程的入度变为 0，则将其加入队列。
5. **检测完成状态**：如果处理完的课程数量等于总课程数量，则可以完成所有课程，否则存在循环依赖。


```cpp
bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
    vector<vector<int>> graph(numCourses);
    vector<int> inDegree(numCourses, 0); // 入度数组

    // 构建图和计算入度
    for (const auto& pre : prerequisites) {
        graph[pre[1]].push_back(pre[0]); // 先修关系
        inDegree[pre[0]]++; // 课程 pre[0] 的入度加 1
    }

    queue<int> q;

    // 将所有入度为 0 的课程加入队列
    for (int i = 0; i < numCourses; ++i) {
        if (inDegree[i] == 0) {
            q.push(i);
        }
    }

    int count = 0; // 记录已处理的课程数量

    // 拓扑排序过程
    while (!q.empty()) {
        int course = q.front();
        q.pop();
        count++; // 处理一门课程

        // 遍历所有后继课程
        for (int next : graph[course]) {
            inDegree[next]--; // 后继课程入度减 1
            if (inDegree[next] == 0) {
                q.push(next); // 如果入度为 0，加入队列
            }
        }
    }

    // 检查是否所有课程都被处理
    return count == numCourses;
}
```
