##### 三个变量滚动
```cpp
class Solution {
   public:
    ListNode* ReverseList(ListNode* head) {
        if (head == nullptr || head->next == nullptr) return head;

        ListNode* pre = head;
        ListNode* mid = head->next;
        ListNode* rea = mid->next;
  
        pre->next=nullptr;
        while (rea != nullptr) {
            mid->next=pre;
            pre=mid;
            mid=rea;
            if (rea->next == nullptr) {
                mid->next=pre;
                break;
            }
            rea=rea->next;
        }
        return rea;
    }
};
```
