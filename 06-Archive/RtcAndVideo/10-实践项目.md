#### 2. 实现WebRTC逻辑

```javascript
// app.js
const localVideo = document.getElementById('localVideo');
const remoteVideo = document.getElementById('remoteVideo');
const startBtn = document.getElementById('startBtn');
const callBtn = document.getElementById('callBtn');
const hangupBtn = document.getElementById('hangupBtn');

let localStream;
let peerConnection;
const ws = new WebSocket('ws://localhost:8080');

// 获取本地媒体流
startBtn.onclick = async () => {
  try {
    localStream = await navigator.mediaDevices.getUserMedia({
      audio: true,
      video: true
    });
    localVideo.srcObject = localStream;
    startBtn.disabled = true;
    callBtn.disabled = false;
  } catch (error) {
    console.error('获取媒体失败:', error);
  }
};

// 创建PeerConnection
function createPeerConnection() {
  const configuration = {
    iceServers: [
      { urls: 'stun:stun.l.google.com:19302' }
    ]
  };
  
  peerConnection = new RTCPeerConnection(configuration);
  
  // 添加本地流
  localStream.getTracks().forEach(track => {
    peerConnection.addTrack(track, localStream);
  });
  
  // 处理远程流
  peerConnection.ontrack = (event) => {
    remoteVideo.srcObject = event.streams[0];
  };
  
  // 处理ICE候选
  peerConnection.onicecandidate = (event) => {
    if (event.candidate) {
      ws.send(JSON.stringify({
        type: 'candidate',
        candidate: event.candidate
      }));
    }
  };
}

// 发起呼叫
callBtn.onclick = async () => {
  createPeerConnection();
  
  // 创建Offer
  const offer = await peerConnection.createOffer();
  await peerConnection.setLocalDescription(offer);
  
  ws.send(JSON.stringify({
    type: 'offer',
    offer: offer
  }));
  
  callBtn.disabled = true;
  hangupBtn.disabled = false;
};

// 挂断
hangupBtn.onclick = () => {
  peerConnection.close();
  peerConnection = null;
  remoteVideo.srcObject = null;
  callBtn.disabled = false;
  hangupBtn.disabled = true;
};

// WebSocket消息处理
ws.onmessage = async (event) => {
  const message = JSON.parse(event.data);
  
  switch(message.type) {
    case 'offer':
      await handleOffer(message.offer);
      break;
    case 'answer':
      await handleAnswer(message.answer);
      break;
    case 'candidate':
      await handleCandidate(message.candidate);
      break;
  }
};

async function handleOffer(offer) {
  if (!peerConnection) {
    createPeerConnection();
  }
  
  await peerConnection.setRemoteDescription(offer);
  const answer = await peerConnection.createAnswer();
  await peerConnection.setLocalDescription(answer);
  
  ws.send(JSON.stringify({
    type: 'answer',
    answer: answer
  }));
}

async function handleAnswer(answer) {
  await peerConnection.setRemoteDescription(answer);
}

async function handleCandidate(candidate) {
  await peerConnection.addIceCandidate(candidate);
}
```

#### 3. 实现信令服务器

```javascript
// server.js
const WebSocket = require('ws');
const wss = new WebSocket.Server({ port: 8080 });

const clients = [];

wss.on('connection', (ws) => {
  clients.push(ws);
  console.log('客户端连接，当前连接数:', clients.length);
  
  ws.on('message', (message) => {
    // 转发给其他客户端
    clients.forEach(client => {
      if (client !== ws && client.readyState === WebSocket.OPEN) {
        client.send(message);
      }
    });
  });
  
  ws.on('close', () => {
    const index = clients.indexOf(ws);
    if (index > -1) {
      clients.splice(index, 1);
    }
    console.log('客户端断开，当前连接数:', clients.length);
  });
});

console.log('信令服务器运行在 ws://localhost:8080');
```

## 10.2 项目二：多人视频会议

### 10.2.1 项目目标

实现一个多人视频会议应用，支持：
- 多人同时视频通话
- 房间管理
- 用户列表显示
- 静音/取消静音
- 开启/关闭摄像头

### 10.2.2 架构设计

使用SFU架构：
- 每个客户端只上传一路流
- 服务器转发给其他客户端
- 客户端接收多路流并显示

### 10.2.3 实现要点

#### 房间管理

```javascript
// 房间管理
class RoomManager {
  constructor() {
    this.rooms = new Map();
  }
  
  createRoom(roomId) {
    if (!this.rooms.has(roomId)) {
      this.rooms.set(roomId, {
        clients: [],
        createdAt: Date.now()
      });
    }
    return this.rooms.get(roomId);
  }
  
  joinRoom(roomId, client) {
    const room = this.createRoom(roomId);
    room.clients.push(client);
    return room;
  }
  
  leaveRoom(roomId, client) {
    const room = this.rooms.get(roomId);
    if (room) {
      const index = room.clients.indexOf(client);
      if (index > -1) {
        room.clients.splice(index, 1);
      }
    }
  }
  
  broadcastToRoom(roomId, message, excludeClient) {
    const room = this.rooms.get(roomId);
    if (room) {
      room.clients.forEach(client => {
        if (client !== excludeClient && client.readyState === WebSocket.OPEN) {
          client.send(JSON.stringify(message));
        }
      });
    }
  }
}
```

#### 多路流管理

```javascript
// 多路流管理
class MultiStreamManager {
  constructor() {
    this.remoteStreams = new Map();
  }
  
  addStream(userId, stream) {
    this.remoteStreams.set(userId, stream);
    this.updateDisplay();
  }
  
  removeStream(userId) {
    this.remoteStreams.delete(userId);
    this.updateDisplay();
  }
  
  updateDisplay() {
    // 更新视频显示
    const container = document.getElementById('remoteVideos');
    container.innerHTML = '';
    
    this.remoteStreams.forEach((stream, userId) => {
      const video = document.createElement('video');
      video.srcObject = stream;
      video.autoplay = true;
      video.id = `video-${userId}`;
      container.appendChild(video);
    });
  }
}
```

## 10.3 项目三：直播推流

### 10.3.1 项目目标

实现一个简单的直播推流应用，包括：
- 主播推流
- 观众观看
- 实时弹幕
- 观看人数统计

### 10.3.2 技术方案

- **推流**：WebRTC推流到服务器
- **分发**：服务器使用SFU或MCU架构
- **播放**：观众通过WebRTC或HLS观看

### 10.3.3 实现要点

#### 推流端

```javascript
// 推流
class StreamPublisher {
  constructor(streamId) {
    this.streamId = streamId;
    this.peerConnection = null;
  }
  
  async startPublishing(localStream) {
    const configuration = {
      iceServers: [
        { urls: 'stun:stun.l.google.com:19302' },
        {
          urls: 'turn:turnserver.com:3478',
          username: 'user',
          credential: 'pass'
        }
      ]
    };
    
    this.peerConnection = new RTCPeerConnection(configuration);
    
    // 添加本地流
    localStream.getTracks().forEach(track => {
      this.peerConnection.addTrack(track, localStream);
    });
    
    // 创建Offer
    const offer = await this.peerConnection.createOffer();
    await this.peerConnection.setLocalDescription(offer);
    
    // 发送给服务器
    this.sendOfferToServer(offer);
  }
  
  sendOfferToServer(offer) {
    fetch('/api/publish', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        streamId: this.streamId,
        offer: offer
      })
    })
    .then(response => response.json())
    .then(data => {
      this.peerConnection.setRemoteDescription(data.answer);
    });
  }
}
```

```javascript
class StreamViewer {
  constructor(streamId) {
    this.streamId = streamId;
    this.peerConnection = null;
  }
  
  async startViewing() {
    const configuration = {
      iceServers: [
        { urls: 'stun:stun.l.google.com:19302' }
      ]
    };
    
    this.peerConnection = new RTCPeerConnection(configuration);
    
    // 处理远程流
    this.peerConnection.ontrack = (event) => {
      const video = document.getElementById('video');
      video.srcObject = event.streams[0];
    };
    
    // 从服务器获取流信息
    const response = await fetch(`/api/watch/${this.streamId}`);
    const data = await response.json();
    
    await this.peerConnection.setRemoteDescription(data.offer);
    const answer = await this.peerConnection.createAnswer();
    await this.peerConnection.setLocalDescription(answer);
    
    // 发送Answer给服务器
    await fetch(`/api/watch/${this.streamId}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ answer: answer })
    });
  }
}
```

```javascript
async function shareScreen() {
  try {
    const screenStream = await navigator.mediaDevices.getDisplayMedia({
      video: true,
      audio: true
    });
    
    // 替换视频轨道
    const videoTrack = screenStream.getVideoTracks()[0];
```

```javascript
class VirtualBackground {
  constructor(videoElement) {
    this.video = videoElement;
    this.canvas = document.createElement('canvas');
    this.ctx = this.canvas.getContext('2d');
    this.backgroundImage = new Image();
  }

    // 设置Canvas尺寸

    // 开始处理
    this.process();
  }

    // 绘制背景
```

```javascript
// 性能监控
class PerformanceMonitor {
  constructor() {
    this.metrics = {
      fps: 0,
      bitrate: 0,
      packetLoss: 0,
      latency: 0
    };
  }
  
  async update() {
    const stats = await peerConnection.getStats();
    
    stats.forEach(report => {
      if (report.type === 'outbound-rtp') {
        this.metrics.bitrate = report.bytesSent * 8 / 1000; // kbps
        this.metrics.packetLoss = report.packetsLost;
      }
      
      if (report.type === 'inbound-rtp') {
        this.metrics.latency = report.jitter;
      }
    });
    
    console.log('性能指标:', this.metrics);
  }
  
  start() {
    setInterval(() => this.update(), 1000);
  }
}
```
